powfacpy.pf_classes.protocol_generator

Create protocol classes for all python classes in the PF python scripting reference pdf file (by running this python module). See output module 'pf_class_protocols' which is used in powfacpy. If you run this module a new module is created ('pf_class_protocols_new'). You can then delete the old one and rename the new file.

IMPORTANT: If you miss a class please add it to the 'missing_classes_that_are_not_in_scripting_reference' list below.

This module is not used by other modules of powfacpy and has the sole purpose of creating the protocols.

116    def get_all_class_names_from_scripting_reference(self) -> list[str]:
117        """Uses a .txt file to which the TOC of the python scripting reference pdf document was copied (includes the class names).
118        Iterates all words from the TOC and checks if they are PF class names.
119
120        Returns:
121            list[str]: PF class names
122        """
123        pf_class_names = []
124        with open(
125            powfacpy_project_root_directory
126            + r"\power_factory_objects\CopyOfObjectsInTableOfContentsOfScriptingReference.txt",
127            "r",
128        ) as in_file:
129            for line in in_file:
130                words = line.split()
131                for word in words:
132                    if self.is_pf_class(word):
133                        pf_class_names.append(word)
134        return pf_class_names

136    def get_pf_objects(self, pf_class_names: list[str]) -> list:
137        """Get instances of the PF classes in 'pf_class_names'.
138
139        Creates a new project and tries to create each class in one of the folders in the project. Some objects can only be created inside certain folders. Hence, the procedure is done twice to ensure that the folders are present (i.e. create folders in first run, use those folders to create the specific objects in the second run).
140        Finally, further objects that cannot be created in this way are added.
141
142        Args:
143            pf_class_names (list[str]): _description_
144
145        Returns:
146            pf_class_objects: list of powerfactory class instances/objects
147            pf_class_objects_not_found: list of objects that could not be found/created
148        """
149        try:
150            self.app.Hide()
151            project = self.app.CreateProject("delete_this_project", "grid")
152            project.Activate()
153
154            for n in range(2):
155                folders = self.get_obj("*", include_subfolders=True)
156                pf_class_objects = []
157                pf_class_objects_not_found = []
158                for pf_class in pf_class_names:
159                    for folder in folders:
160                        pf_class_obj = None
161                        try:
162                            pf_class_obj = self.create_in_folder(
163                                "new." + pf_class, folder
164                            )
165                            if pf_class_obj:
166                                break  # break loop when successfully created an object
167                        except RuntimeError:  # CreateObject can raise runtime error
168                            pass
169                    if not pf_class_obj:
170                        pf_class_objects_not_found.append(pf_class)
171                    else:
172                        pf_class_objects.append(pf_class_obj)
173            # Insert some further objects
174            pf_class_objects.insert(0, self.app)
175            pf_class_objects.append(project)
176            pf_class_objects_not_found.remove(pf_class_objects[-1].GetClassName())
177            pf_class_objects.append(self.get_active_user_folder())
178            pf_class_objects_not_found.remove(pf_class_objects[-1].GetClassName())
179            pf_class_objects.append(self.get_first_level_folder("global library"))
180            pf_class_objects_not_found.remove(pf_class_objects[-1].GetClassName())
181        finally:
182            self.app.Show()
183        return pf_class_objects, pf_class_objects_not_found, project

185    def get_class_definition(self, pf_obj) -> str:
186        """Get string with the definition of a protocol class for a PF object.
187        Example: 'class ElmTr2(Protocol):'
188
189        Args:
190            pf_obj: PF object
191
192        Returns:
193            str: class definition
194        """
195        definition: str = ""
196        if not isinstance(pf_obj, type(self.app)):
197            class_name = pf_obj.GetClassName()
198            definition += "class " + class_name + "(PFGeneral):\n"
199        else:
200            class_name = "PFApp"
201            definition += "class " + class_name + "(Protocol):\n"
202
203        if hasattr(pf_obj, "GetClassDescription"):
204            class_description = self.app.GetClassDescription(
205                class_name
206            )  # It seems that GetClassDescription always retuns an empty string for every class
207            if class_description:
208                definition += (
209                    self.indentation_increment + '"""' + class_description + '"""\n\n'
210                )
211        return definition

213    def get_class_attribute_definitions(self, pf_obj) -> str:
214        """Get definitions of class attributes (data attributes and methods) of a PF object.
215
216        Args:
217            pf_obj: PF object
218
219        Returns:
220            str: definitions of all attributes of the class of the object
221        """
222        class_attributes: set = (
223            set(dir(pf_obj)) - set(dir(Empty)) - self.general_methods
224        )  # omit attributes that all python objects have and general methods.
225        class_attributes = sorted(class_attributes)
226        data_attributes = ""
227        methods = ""
228        for attr in class_attributes:
229            # In rare cases, the attributes are named after python keywords - such names are not allowed in python classes and are omitted.
230            if hasattr(pf_obj, attr) and not keyword.iskeyword(attr):
231                if callable(getattr(pf_obj, attr)):
232                    # Due to lack of knowledge about the method arguments, the generic '(*args)' is used.
233                    methods += (
234                        self.indentation_increment
235                        + "def "
236                        + attr
237                        + "(*args):\n"
238                        + 2 * self.indentation_increment
239                        + "...\n\n"
240                    )
241                else:
242                    attribute_value = getattr(pf_obj, attr)
243                    attribute_type = type(attribute_value).__name__
244                    # The types "DataObject", "NoneType" are replace with 'object'
245                    # "DataObject" is replaced because powfacpy should not depend on an import of the powerfactory module (which is necessary for this type hint)
246                    # "NoneType" is not a helpful information
247                    if not attribute_type in ["DataObject", "NoneType"]:
248                        data_attributes += (
249                            self.indentation_increment
250                            + attr
251                            + ": "
252                            + attribute_type
253                            + "\n"
254                        )
255                    else:
256                        data_attributes += (
257                            self.indentation_increment + attr + ": object\n"
258                        )
259                    if not isinstance(pf_obj, type(self.app)):
260                        attribute_description = pf_obj.GetAttributeDescription(attr)
261                        if attribute_description:
262                            # some characters need to be deleted (newlines, '\xb0' is not a unicode symbol)
263                            attribute_description = (
264                                attribute_description.replace("\n", "")
265                                .replace('"', "")
266                                .replace("\xb0", "")
267                            )
268                            data_attributes += (
269                                self.indentation_increment
270                                + '"'
271                                + attribute_description
272                                + '"\n'
273                            )
274        return data_attributes, methods

285                + 2 * self.indentation_increment

287            )
288        return class_str

137    def get_pf_objects(self, pf_class_names: list[str]) -> list:
138        """Get instances of the PF classes in 'pf_class_names'.
139
140        Creates a new project and tries to create each class in one of the folders in the project. Some objects can only be created inside certain folders. Hence, the procedure is done twice to ensure that the folders are present (i.e. create folders in first run, use those folders to create the specific objects in the second run).
141        Finally, further objects that cannot be created in this way are added.
142
143        Args:
144            pf_class_names (list[str]): _description_
145
146        Returns:
147            pf_class_objects: list of powerfactory class instances/objects
148            pf_class_objects_not_found: list of objects that could not be found/created
149        """
150        try:
151            self.app.Hide()
152            project = self.app.CreateProject("delete_this_project", "grid")
153            project.Activate()
154
155            for n in range(2):
156                folders = self.get_obj("*", include_subfolders=True)
157                pf_class_objects = []
158                pf_class_objects_not_found = []
159                for pf_class in pf_class_names:
160                    for folder in folders:
161                        pf_class_obj = None
162                        try:
163                            pf_class_obj = self.create_in_folder(
164                                "new." + pf_class, folder
165                            )
166                            if pf_class_obj:
167                                break  # break loop when successfully created an object
168                        except RuntimeError:  # CreateObject can raise runtime error
169                            pass
170                    if not pf_class_obj:
171                        pf_class_objects_not_found.append(pf_class)
172                    else:
173                        pf_class_objects.append(pf_class_obj)
174            # Insert some further objects
175            pf_class_objects.insert(0, self.app)
176            pf_class_objects.append(project)
177            pf_class_objects_not_found.remove(pf_class_objects[-1].GetClassName())
178            pf_class_objects.append(self.get_active_user_folder())
179            pf_class_objects_not_found.remove(pf_class_objects[-1].GetClassName())
180            pf_class_objects.append(self.get_first_level_folder("global library"))
181            pf_class_objects_not_found.remove(pf_class_objects[-1].GetClassName())
182        finally:
183            self.app.Show()

def get_class_attribute_definitions(self, pf_obj) -> str: 
214    def get_class_attribute_definitions(self, pf_obj) -> str:
215        """Get definitions of class attributes (data attributes and methods) of a PF object.
216
217        Args:
218            pf_obj: PF object
219
220        Returns:
221            str: definitions of all attributes of the class of the object
222        """
223        class_attributes: set = (
224            set(dir(pf_obj)) - set(dir(Empty)) - self.general_methods
225        )  # omit attributes that all python objects have and general methods.
226        class_attributes = sorted(class_attributes)
227        data_attributes = ""
228        methods = ""
229        for attr in class_attributes:
230            # In rare cases, the attributes are named after python keywords - such names are not allowed in python classes and are omitted.
231            if hasattr(pf_obj, attr) and not keyword.iskeyword(attr):
232                if callable(getattr(pf_obj, attr)):
233                    # Due to lack of knowledge about the method arguments, the generic '(*args)' is used.
234                    methods += (
235                        self.indentation_increment
236                        + "def "
237                        + attr
238                        + "(*args):\n"
239                        + 2 * self.indentation_increment
240                        + "...\n\n"
241                    )
242                else:
243                    attribute_value = getattr(pf_obj, attr)
244                    attribute_type = type(attribute_value).__name__
245                    # The types "DataObject", "NoneType" are replace with 'object'
246                    # "DataObject" is replaced because powfacpy should not depend on an import of the powerfactory module (which is necessary for this type hint)
247                    # "NoneType" is not a helpful information
248                    if not attribute_type in ["DataObject", "NoneType"]:
249                        data_attributes += (
250                            self.indentation_increment
251                            + attr
252                            + ": "
253                            + attribute_type
254                            + "\n"
255                        )
256                    else:
257                        data_attributes += (
258                            self.indentation_increment + attr + ": object\n"
259                        )
260                    if not isinstance(pf_obj, type(self.app)):
261                        attribute_description = pf_obj.GetAttributeDescription(attr)
262                        if attribute_description:
263                            # some characters need to be deleted (newlines, '\xb0' is not a unicode symbol)
264                            attribute_description = (
265                                attribute_description.replace("\n", "")
266                                .replace('"', "")
267                                .replace("\xb0", "")
268                            )
269                            data_attributes += (
270                                self.indentation_increment
271                                + '"'
272                                + attribute_description
273                                + '"\n'
274                            )
275        return data_attributes, methods
