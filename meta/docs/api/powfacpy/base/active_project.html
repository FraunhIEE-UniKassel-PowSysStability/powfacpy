 139        self,

 144        """Set attributes of an object and to store the original values (e.g. to reset them later).

 196    def get_from_study_case(
 197        self,
 198        class_name: str,
 199        if_not_unique: str = "warning",
 200        if_no_study_case: str = "error",
 201    ) -> PFGeneral:
 202        """Get objects from active study case (similar to PF built-in function 'app.GetFromStudyCase()').
 203
 204        Additionally, this method prints a warning or raises an exception if there is more than one object found in the study case and if no study case is activated.
 205
 206        Args:
 207            class_name (str): class name of the object (e.g. 'ElmRes'), optionally preceded by an object name without wildcards and a dot (e.g. 'All Calcualations.ElmRes')
 208
 209            if_not_unique (str, optional): Warn ('warning') or raise exception ('error') if there are more than one objects of class 'class_name'. Defaults to "warning".
 210
 211            if_no_study_case (str, optional): Warn ('warning') or raise exception ('error') if no study case is active. Defaults to "error".
 212
 213        Raises:
 214            PFNoActiveStudyCaseError: No study case activated
 215            TypeError: More than one object was found
 216
 217        Returns:
 218            PFGeneral: Found or created object
 219        """
 220        obj = self.__class__.app.GetFromStudyCase(class_name)
 221
 222        if if_no_study_case and not self.__class__.app.GetActiveStudyCase():
 223            if if_no_study_case == "warning":
 224                warn(
 225                    "No study case activated. PowerFactory creates object of class_name in tmp folder, outside any study case."
 226                )
 227            elif if_no_study_case == "error":
 228                raise PFNoActiveStudyCaseError()
 229
 230        if if_not_unique and self.is_pf_class(class_name):
 231            class_name = "*." + class_name
 232            all_objects_of_this_class = self.get_obj(
 233                class_name, parent_folder=obj.GetParent(), include_subfolders=False
 234            )
 235            if len(all_objects_of_this_class) > 1:
 236                parent_path = self.get_path_of_object(obj.GetParent())
 237                if if_not_unique == "warning":
 238                    warn(
 239                        f"The returned {class_name} object is not unique in the  study case: '{parent_path}'. Make sure that the correct {class_name} object is used: {obj}."
 240                    )
 241                if if_not_unique == "error":
 242                    raise TypeError(
 243                        f"The returned {class_name} object is not unique in its folder / in its study case: '{parent_path}'."
 244                    )
 245        return obj

 539    def create_study_case(
 540        self,
 541        name: str,
 542        copy_from: IntCase | str | None = None,
 543        parent_folder: PFGeneral | str | None = None,
 544        create_variation: bool = False,
 545        create_scenario: bool = False,
 546        overwrite: bool = True,
 547        use_existing=False,
 548        activate: bool = True,
 549    ) -> IntCase | list[IntCase | IntScheme | IntScenario]:
 550        """Create a new study case and optionally a variation and/or scenario.
 551
 552        Args:
 553            name (str): name (used for case name and variation/scenario name)
 554            copy_from (IntCase | str | None, optional): case to copy from. Defaults to None.
 555            parent_folder (PFGeneral | str | None, optional): parent folder (same subfolders are used for variation/scenario). Defaults to None.
 556            create_variation (bool, optional): Defaults to False.
 557            create_scenario (bool, optional): Defaults to False.
 558            overwrite (bool, optional): existing objects are overwritten. Defaults to True.
 559            use_existing (bool, optional): existing objects are used. Defaults to False.
 560            activate (bool, optional): activate case/variation/scenario. Defaults to True.
 561
 562        Returns:
 563            IntCase | list[IntCase | IntScheme | IntScenario]: Created case or list with case/variation/scenario.
 564        """
 565        if parent_folder is None:
 566            parent_folder = self.study_cases_folder
 567        elif isinstance(parent_folder, str):
 568            if "\\" in parent_folder:
 569                parent_folder = self.create_by_path(
 570                    parent_folder + ".IntFolder", overwrite=False, use_existing=True
 571                )
 572            else:
 573                parent_folder = self.study_cases_folder
 574        if copy_from is None:
 575            case = self.create_in_folder(
 576                name + ".IntCase",
 577                parent_folder,
 578                overwrite=overwrite,
 579                use_existing=use_existing,
 580            )
 581        else:
 582            case = self.copy_single_obj(
 583                copy_from,
 584                target_folder=parent_folder,
 585                new_name=name,
 586                overwrite=overwrite,
 587                use_existing=use_existing,
 588            )
 589        if activate:
 590            case.Activate()
 591        if not create_variation and not create_scenario:
 592            return case
 593        else:

540    def create_study_case(
541        self,
542        name: str,
543        copy_from: IntCase | str | None = None,
544        parent_folder: PFGeneral | str | None = None,
545        create_variation: bool = False,
546        create_scenario: bool = False,
547        overwrite: bool = True,
548        use_existing=False,
549        activate: bool = True,
550    ) -> IntCase | list[IntCase | IntScheme | IntScenario]:
551        """Create a new study case and optionally a variation and/or scenario.
552
553        Args:
554            name (str): name (used for case name and variation/scenario name)
555            copy_from (IntCase | str | None, optional): case to copy from. Defaults to None.
556            parent_folder (PFGeneral | str | None, optional): parent folder (same subfolders are used for variation/scenario). Defaults to None.
557            create_variation (bool, optional): Defaults to False.
558            create_scenario (bool, optional): Defaults to False.
559            overwrite (bool, optional): existing objects are overwritten. Defaults to True.
560            use_existing (bool, optional): existing objects are used. Defaults to False.
561            activate (bool, optional): activate case/variation/scenario. Defaults to True.
562
563        Returns:
564            IntCase | list[IntCase | IntScheme | IntScenario]: Created case or list with case/variation/scenario.
565        """

567            parent_folder = self.study_cases_folder
568        elif isinstance(parent_folder, str):
569            if "\\" in parent_folder:
570                parent_folder = self.create_by_path(
571                    parent_folder + ".IntFolder", overwrite=False, use_existing=True
572                )
573            else:
574                parent_folder = self.study_cases_folder
575        if copy_from is None:
576            case = self.create_in_folder(
577                name + ".IntCase",
578                parent_folder,
579                overwrite=overwrite,

583            case = self.copy_single_obj(
584                copy_from,

586                new_name=name,
587                overwrite=overwrite,
588                use_existing=use_existing,

594        else:
595            returned_objs = [case]
596        if create_variation:
597            returned_objs.append(
598                self.create_parallel_variation_for_study_case(
599                    case, overwrite=overwrite, activate=activate

602        if create_scenario:

606                )
607            )
608        return returned_objs
