 87    def get_load_flow_state(
 88        self, execute_load_flow: bool = True, format: str | None = "pandas"
 89    ) -> None:
 90        """Get load flow results of subsystem.
 91
 92        Includes total load/generation, power exchange etc.
 93
 94        Args:
 95            execute_load_flow (bool, optional): _description_. Defaults to True.
 96            format (str | None, optional): Format of the returned results. Options are 'pandas', 'dict' or None. Defaults to "pandas".
 97        """
 98        if execute_load_flow:
 99            self.parent.act_prj.execute_load_flow()
100        self.total_power_loads = self.parent.load_flow_total_power_loads_in_MVA()
101        self.total_power_generation = (
102            self.parent.load_flow_total_power_generation_in_MVA()
103        )
104        self.total_power_exchange = self.parent.load_flow_total_power_exchange_in_MVA()
105        if format is None:
106            return None
107        elif format == "pandas":
108            return pd.DataFrame(
109                {
110                    "total_power_loads": self.total_power_loads,
111                    "total_power_generation": self.total_power_generation,
112                    "total_power_exchange": self.total_power_exchange,
113                },
114                index=[self.parent.name],
115            )
116        elif format == "dict":
117            return {
118                "total_power_loads": self.total_power_loads,
119                "total_power_generation": self.total_power_generation,
120                "total_power_exchange": self.total_power_exchange,
121            }

309    def delete_subsystem(self, subsystem: int | SubSystem) -> None:
310        """Delete subsystem from the container.
311
312        Ensure that all attributes of the container that depend on the subsystems are updated/reset accordingly (e.g. power exchange matrix, total loads/generation etc.).
313
314        Args:
315            subsystem (int | SubSystem): subsystem or its index in the container to be deleted

343    def get_power_exchange_in_MVA(
344        self,
345        execute_load_flow: bool = True,
346        format: str = "dataframe",
347        value_no_exchange=np.nan + 0j,
348    ) -> np.array | pd.DataFrame:
349        """Get power exchange between subsystems in MVA.
350
351        Args:
352            execute_load_flow (bool, optional): If true, execute load flow first to ensure that results are present. Defaults to True.
353            format (str, optional): If "dataframe", return a pandas DataFrame; otherwise, return a numpy array. Defaults to "dataframe".
354            value_no_exchange (_type_, optional): The value to use for no exchange between subsystems that are not connected. Defaults to np.nan+0j.
355
356        Returns:
357            np.array | pd.DataFrame: power exchange.
358        """
359        if execute_load_flow:
360            self.act_prj.execute_load_flow()
361        s_exchange = np.full(
362            (self.num_subsystems, self.num_subsystems), value_no_exchange
363        )
364        for m, z1 in enumerate(self.elm_zones):
365            for n, z2 in enumerate(self.elm_zones[m + 1 :]):
366                if z1.CalculateInterchangeTo(z2) > 0:
367                    s_exchange[m, m + n + 1] = z1.GetAttribute(
368                        "c:Pinter"
369                    ) + 1j * z1.GetAttribute("c:Qinter")
370                    s_exchange[m + n + 1, m] = -s_exchange[m, m + n + 1]
371        labels = self.names
372        self._power_exchange = pd.DataFrame(s_exchange, index=labels, columns=labels)
373        self._attributes_depending_on_all_subsystems.append("_power_exchange")
374        if format == "dataframe":
375            return self._power_exchange
376        else:
377            return s_exchange

379    def get_neighboring_subsystems_in_container(
380        self, subsystem: int | SubSystem | ElmZone
381    ) -> list[SubSystem]:
382        """Get neighbors of 'subsystem' in container.
383
384        Args:
385            subsystem (int | SubSystem): subsystem or its index in the container for which the neighbors should be found.
386
387        Returns:
388            list[SubSystem]: List of neighboring subsystems.
389        """
390        subsystem = self._handle_subsystem_input(subsystem)
391        self.act_prj.check_load_flow_results(when_invalid="execute")
392        return [
393            self._subsystems[n]
394            for n, subs in enumerate(self._subsystems)
395            if not subs == subsystem
396            and subsystem._obj.CalculateInterchangeTo(subs._obj) > 0
397        ]

411        if execute_load_flow:

414            [
415                subs.get_load_flow_state(execute_load_flow=False, format="pandas")

345        self,
346        execute_load_flow: bool = True,
347        format: str = "dataframe",
348        value_no_exchange=np.nan + 0j,
349    ) -> np.array | pd.DataFrame:

351

369                        "c:Pinter"
370                    ) + 1j * z1.GetAttribute("c:Qinter")
371                    s_exchange[m + n + 1, m] = -s_exchange[m, m + n + 1]
372        labels = self.names
373        self._power_exchange = pd.DataFrame(s_exchange, index=labels, columns=labels)
374        self._attributes_depending_on_all_subsystems.append("_power_exchange")
