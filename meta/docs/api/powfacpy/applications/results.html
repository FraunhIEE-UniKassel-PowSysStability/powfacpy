207    def _format_csv_for_comtrade(self, file_path: str) -> None:
208        """Format the .csv file created (using ComRes) based on a Comtrade object (IntComtrade).
209        There is a bug in PF so that the time in the first column sometimes
210        is not monotonously increasing. This methods corrects this by checking
211        for each time value (1. column) whether it is larger than the previous and discarding rows
212        where this is not the case.
213        """
214        with open(file_path) as read_file, open(file_path + ".temp", "w") as write_file:
215            row = read_file.readline()
216            write_file.write(row)
217            time = inf
218            row = read_file.readline()
219            while row:
220                row_entries = row.split(",")
221                if float(row_entries[0]) > time:
222                    write_file.write(row)
223                    time = float(row_entries[0])
224                row = read_file.readline()
225        replace(file_path + ".temp", file_path)

227    def _format_csv_for_elmres(
228        self, file_path: str, num_header_rows: int, list_of_results_objs: list
229    ) -> None:
230        """Format the csv file that is exported from PF.
231        The PF exported csv uses the first rows either for
232        - result object name (e.g. ElmRes name), element name (e.g. ElmTerm), variable name (e.g. "m:u")
233
234        or only for
235
236        - element name (e.g. ElmTerm), variable name (e.g. "m:u")
237        This must be specified in num_header_rows (either 2 or 3).
238
239        The formatted csv file will use only the first row as a header.
240        This row contains the path of the object and the variable name
241        without description.
242
243        Example first row of some column before formatting:
244          '\\username.IntUser\\powfacpy_base.IntPrj\\Network Model.IntPrjfolder\\Network Data.IntPrjfolder\\Grid.ElmNet\\AC Voltage Source.ElmVac\\s:u0'
245        Example input second row of the column before formatting:
246          's:u0 in kV'
247        Example first row of the column after formatting:
248          'Network Model\\Network Data\\Grid\\AC Voltage Source\\s:u0'
249        """
250        with open(file_path) as read_file, open(file_path + ".temp", "w") as write_file:
251            if num_header_rows == 2:
252                full_paths = read_file.readline().split(",")
253                variables = read_file.readline().split(",")
254            else:
255                read_file.readline().split(",")
256                full_paths = read_file.readline().split(",")
257                variables = read_file.readline().split(",")
258            for col, path in enumerate(full_paths):
259                is_last_column = col == len(full_paths) - 1
260                if col > 0:
261                    path = self._format_path_of_obj_inside_active_project(
262                        PFStringManipulation.format_full_path(path, self.act_prj.app)
263                    )
264                    variable_name = PFStringManipulation.format_variable_name(
265                        variables[col]
266                    )
267                    row = (
268                        row + path + "\\" + variable_name + "," * (not is_last_column)
269                    )  # consistently add headers to row
270                else:
271                    row = "time,"  # Header of first column
272            write_file.write(row + "\n")
273            # Write remaining data rows until end of file is reached
274            while row:
275                row = read_file.readline()
276                write_file.write(row)
277        replace(file_path + ".temp", file_path)
278        if list_of_results_objs:
279            if (
280                len(list_of_results_objs) != len(full_paths) - 1
281            ):  # -1 because of time in first column
282                warn(
283                    "Not all specified results were exported. Some of the elements may be 'out of service' and were not included."
284                )

385        self,
386        df: pd.DataFrame,

413                        col[num_header_rows - 2], self.act_prj.app

208    def _format_csv_for_comtrade(self, file_path: str) -> None:
209        """Format the .csv file created (using ComRes) based on a Comtrade object (IntComtrade).
210        There is a bug in PF so that the time in the first column sometimes
211        is not monotonously increasing. This methods corrects this by checking
212        for each time value (1. column) whether it is larger than the previous and discarding rows
213        where this is not the case.
214        """
215        with open(file_path) as read_file, open(file_path + ".temp", "w") as write_file:
216            row = read_file.readline()
217            write_file.write(row)
218            time = inf
219            row = read_file.readline()
220            while row:
221                row_entries = row.split(",")
222                if float(row_entries[0]) > time:
223                    write_file.write(row)
224                    time = float(row_entries[0])
225                row = read_file.readline()
226        replace(file_path + ".temp", file_path)

228    def _format_csv_for_elmres(
229        self, file_path: str, num_header_rows: int, list_of_results_objs: list
230    ) -> None:
231        """Format the csv file that is exported from PF.
232        The PF exported csv uses the first rows either for
233        - result object name (e.g. ElmRes name), element name (e.g. ElmTerm), variable name (e.g. "m:u")
234
235        or only for
236
237        - element name (e.g. ElmTerm), variable name (e.g. "m:u")
238        This must be specified in num_header_rows (either 2 or 3).
239
240        The formatted csv file will use only the first row as a header.
241        This row contains the path of the object and the variable name
242        without description.
243
244        Example first row of some column before formatting:
245          '\\username.IntUser\\powfacpy_base.IntPrj\\Network Model.IntPrjfolder\\Network Data.IntPrjfolder\\Grid.ElmNet\\AC Voltage Source.ElmVac\\s:u0'
246        Example input second row of the column before formatting:
247          's:u0 in kV'
248        Example first row of the column after formatting:
249          'Network Model\\Network Data\\Grid\\AC Voltage Source\\s:u0'
250        """
251        with open(file_path) as read_file, open(file_path + ".temp", "w") as write_file:
252            if num_header_rows == 2:
253                full_paths = read_file.readline().split(",")
254                variables = read_file.readline().split(",")
255            else:
256                read_file.readline().split(",")
257                full_paths = read_file.readline().split(",")
258                variables = read_file.readline().split(",")
259            for col, path in enumerate(full_paths):
260                is_last_column = col == len(full_paths) - 1
261                if col > 0:
262                    path = self._format_path_of_obj_inside_active_project(
263                        PFStringManipulation.format_full_path(path, self.act_prj.app)
264                    )
265                    variable_name = PFStringManipulation.format_variable_name(
266                        variables[col]
267                    )
268                    row = (
269                        row + path + "\\" + variable_name + "," * (not is_last_column)
270                    )  # consistently add headers to row
271                else:
272                    row = "time,"  # Header of first column
273            write_file.write(row + "\n")
274            # Write remaining data rows until end of file is reached
275            while row:
276                row = read_file.readline()
277                write_file.write(row)
278        replace(file_path + ".temp", file_path)
279        if list_of_results_objs:
280            if (
281                len(list_of_results_objs) != len(full_paths) - 1
282            ):  # -1 because of time in first column
283                warn(
284                    "Not all specified results were exported. Some of the elements may be 'out of service' and were not included."
285                )
