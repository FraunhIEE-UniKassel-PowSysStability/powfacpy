 29    def pf_project_to_pandapower(
 30        self, project_path_in_user: str = None, json_path: str = None
 31    ) -> pp.auxiliary.pandapowerNet:
 32        """
 33        Convert PowerFactory network to pandapower format.
 34
 35        see pandapower docs: https://pandapower.readthedocs.io//en//v2.13.1//converter//powerfactory.html
 36
 37        The method 'from_pfd' that is used deactivates the currently active PF project, so it is reactivated.
 38
 39        Arguments:
 40            project_path_in_user (str): path of the project in PowerFactory database relative to user
 41
 42            json_path (str): path of exported pandapower json file
 43
 44        Returns:
 45            pp.auxiliary.pandapowerNet: pandapower network
 46        """
 47        active_project = self.act_prj.app.GetActiveProject()
 48        if project_path_in_user is None:
 49            project_path_in_user = self.act_prj.get_path_of_object_in_current_user(
 50                active_project
 51            )
 52        net = pp.converter.powerfactory.from_pfd(
 53            self.app, prj_name=project_path_in_user, path_dst=json_path
 54        )
 55        if active_project:
 56            active_project.Activate()
 57        return net

 59    def get_Ybus_matrix(self, net: pp.pandapowerNet, return_deepcopy=True) -> np.matrix:
 60        """
 61        Get admittance matrix (sparse).
 62
 63        If internal bus-branch model does not exist yet, a power flow must be executed.
 64
 65        Arguments:
 66            net: pandapower network
 67            return_deepcopy: If True, a deepcopy of the matrix stored in 'net' is returned.
 68
 69        Returns:
 70            np.matrix: admittance matrix
 71        """
 72        if not net._ppc or net._ppc["internal"]["Ybus"].size == 0:
 73            pp.runpp(net)
 74        # Due to bug
 75        Y_bus = np.array(copy.deepcopy(net._ppc["internal"]["Ybus"]).todense())
 76        np.fill_diagonal(Y_bus, 0)
 77        Y_bus = Y_bus - np.diag(np.sum(Y_bus, axis=0).reshape(-1))
 78        return Y_bus
 79        # if return_deepcopy:
 80        #     return copy.deepcopy(net._ppc["internal"]["Ybus"])
 81        # else:
 82        #     return net._ppc["internal"]["Ybus"]

100    def rearrange_Ybus_frame(
101        self, y_bus: pd.DataFrame, bus_objs: ElmTerm
102    ) -> pd.DataFrame:
103        """Rearrange admittance matrix dataframe in order of 'bus_obj'.
104
105        Args:
106            y_bus (pd.DataFrame): Admittance matrix frame (labels are terminal names)
107            bus_objs (ElmTerm): Terminals in new order.
108
109        Returns:
110            pd.DataFrame: Rearranged admittance matrix frame.
111        """
112        pdi = PandasInterface()
113        columns = copy.deepcopy(y_bus.columns)
114        y_bus = pdi.replace_loc_name_with_pf_objects_in_labels(
115            y_bus,
116            "ElmTerm",
117            index_and_column_labels_are_equal=True,
118        )
119        order = [
120            int(np.where(y_bus.columns.get_level_values(0) == bus)[0])
121            for bus in bus_objs
122        ]
123        y_bus.iloc[:, :] = rearrange_matrix_rows_and_cols(y_bus.to_numpy(), order)
124        y_bus.columns = columns[order]
125        y_bus.index = y_bus.columns
126        return y_bus

145    def get_admittance_matrix_labels(self, net: pp.pandapowerNet):
146        matrix_index_labels = [None] * net._ppc["internal"]["Ybus"].shape[0]

149            matrix_index_labels[ybus_idx] = net.bus.iloc[bus_idx, :]["name"]

184    def get_connectivity_matrix(
185        self, net: pp.pandapowerNet, boolean: bool = False
186    ) -> np.array:
187        """Get the connectivity (also called adjacency) matrix.
188
189        Indices of connected nodes are True. The connectivity matrix is the admittance matrix where nonzero entries are True, other entries are False.
190
191        Args:
192            net (pp.pandapowerNet): pandapower dataset
193
194        Returns:
195            np.array[bool]: Connectivity matrix.
196        """
197        Ybus = net._ppc["internal"]["Ybus"]
198        if not boolean:
199            connect_mat = np.full(Ybus.shape, 0)
200            connect_mat[np.array(Ybus.todense() != 0)] = 1
201        else:
202            connect_mat = np.full(Ybus.shape, False)
203            connect_mat[np.array(Ybus.todense() != 0)] = True
204        return connect_mat

215    def get_difference_between_pf_and_pandapower_dataset(
216        self,
217        net: pp.auxiliary.pandapowerNet,
218    ) -> pd.DataFrame | None:
219        """
220        Get difference between pandapower dataset (exported from PF) and the original PF.
221
222        Assumes default settings for units in the PF project (see Settings/Units). Note that only a small subset of classes and parameters is checked (see 'get_pandapower_2_pf_parameter_mapping' and 'get_pandapower_2_pf_class_mapping')
223
224        Args:
225            net (pp.auxiliary.pandapowerNet): pandapower dataset (exported from PF)
226
227        Returns:
228            pd.DataFrame | None: Dataframe with differences. None if there are no differences
229        """
230        pandapower_2_pf_class_mapping = self.get_pandapower_2_pf_class_mapping()
231        pandapower_2_pf_parameter_mapping = self.get_pandapower_2_pf_parameter_mapping()
232        differing_objects = {
233            "name": [],
234            "pf_class": [],
235            "pp_parameter": [],
236            "pf_value": [],
237            "pp_value": [],
238        }
239        for pp_class, pf_classes in pandapower_2_pf_class_mapping.items():
240            df_pp_class = net[pp_class].set_index("name")
241            for pf_class in pf_classes:
242                pf_objects: list[PFGeneral] = self.app.GetCalcRelevantObjects(
243                    f"*.{pf_class}"
244                )
245                for pp_param, pf_param in pandapower_2_pf_parameter_mapping[
246                    pf_class
247                ].items():
248                    for pf_obj in pf_objects:
249                        if isinstance(pf_param, str):
250                            pf_value = getattr(pf_obj, pf_param)
251                        else:
252                            pf_value = pf_param(pf_obj)
253                        pp_value = df_pp_class.loc[pf_obj.loc_name, pp_param]
254                        if not (pf_value == pp_value):
255                            differing_objects["name"].append(pf_obj.loc_name)
256                            differing_objects["pf_class"].append(pf_obj.GetClassName())
257                            differing_objects["pp_parameter"].append(pp_param)
258                            differing_objects["pf_value"].append(pf_value)
259                            differing_objects["pp_value"].append(pp_value)
260        if not differing_objects["name"]:
261            return None
262        else:
263            return pd.DataFrame(differing_objects)
