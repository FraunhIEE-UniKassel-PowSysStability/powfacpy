 86    def get_name_for_each_step(self, use_calculated_vales: bool = False) -> list[str]:
 87        """Get a list of names for each steps. The names consist of 'self.name' and the value of the step (if the values of the targeted PF objects differ, the value of the first object in 'self.values' is used).
 88
 89        Args:
 90            use_calculated_vales (bool): If True, the calculated values are used, else the values as specified by the user are used. Relevant if self.value_calculation is 'multiply' or 'add. Defaults to False.
 91
 92        Returns:
 93            list[str]: list of names
 94        """
 95        if self.value_calculation in ["multiply", "add"] and not use_calculated_vales:
 96            # To get the values as specified by the user, the values are calculated as if 'self.value_calculation = "equal"'
 97            value_calculation = self.value_calculation
 98            self.value_calculation = "equal"
 99            values = list(self.get_values().values())[0]
100            self.value_calculation = value_calculation
101        else:
102            values = list(self.values.values())[0]
103        return [
104            f"{self.name}={format_sig_no_sci(val)}" for val in values
105        ]  # Format numerics

107    def get_values(self) -> dict[PFGeneral, Any]:
108        """Get the values of each PF object for each step.
109
110        Returns:
111            dict[PFGeneral, Any]: PF objects, attributes (keys) and list of values (values)
112        """
113        if self.variation_type == "const":  # values for all objects are the same
114            values = np.broadcast_to(
115                np.linspace(self.min, self.max, self.steps),
116                (self.number_of_objs, self.steps),
117            )
118        elif self.variation_type == "gradient":  # calculate gradients
119            grad_start_objs = np.linspace(self.min, self.max, self.number_of_objs)
120            values = np.array(
121                [
122                    np.linspace(grad_start_objs[n], grad_start_objs[-n - 1], self.steps)
123                    for n in range(len(grad_start_objs))
124                ]
125            )
126        # create dict (keys are obj, attr and values are their values at each step)
127        if isinstance(self.objs_and_attr, list):
128            values = {
129                (obj, self.objs_and_attr[1]): values[n, :]
130                for n, obj in enumerate(self.objs_and_attr[0])
131            }
132        else:
133            values = {
134                (obj, attr): values[n, :]
135                for n, (obj, attr) in enumerate(self.objs_and_attr.items())
136            }
137        # multiply or add to default values
138        if self.value_calculation == "multiply":
139            values = {k: v * self.default_values[k] for k, v in values.items()}
140        elif self.value_calculation == "add":
141            values = {k: v + self.default_values[k] for k, v in values.items()}
142        return values

280    def get_all_values(self) -> dict:
281        """Get values of PF object attributes for all variants.
282
283        Returns:
284            dict: Tuple of PF object, attribute (keys) and (flat) list of attribute values for all variants (values).
285        """
286        all_values = {k: [] for k in self.get_all_objs_and_attr()}
287        for obj_and_attr, values in all_values.items():
288            for var in self.variants:
289                val = var.values.get(obj_and_attr)
290                if val is not None:
291                    values += list(val)
292                else:  # use default value
293                    values += [self.default_values[obj_and_attr]] * var.steps
294        all_values = {k: np.array(v) for k, v in all_values.items()}
295        return all_values

297    def step_generator(
298        self, set_values_in_pf: bool = True
299    ) -> Generator[tuple[dict, str]]:
300        for variant in self.variants:
301            for step_num, name in enumerate(variant.names_for_each_value):
302                values = {}
303                for (obj, attr), vals in variant.values.items():
304                    value = vals[step_num]
305                    values[(obj, attr)] = value
306                    if set_values_in_pf and not isinstance(obj, str):
307                        obj.SetAttribute(attr, value)
308                yield values, name

310    def execute_dynamic_simulations(
311        self,
312    ) -> dict:
313        """Execute dynamic simulations for all steps of all variants. Allows functions calls for pre- and postprocessing (the functions take all local variables as input arguments).
314
315        Args:
316            sample_time (float): resample simulation results
317
318        Returns:
319            dict: simulation result DataFrames (values) of all simulations (keys are names of simulations)
320        """
321        pfds = DynamicSimulation()
322        pfri = Results()
323        pfri.pf_objects_in_labels = True
324        all_sim_res = {}
325        sim_num = 0
326        self.set_default_values()
327        for variant in self.variants:
328            try:
329                self.execute_preprocessing_variant(locals())
330                for step_num, name in enumerate(variant.names_for_each_value):
331                    self.execute_preprocessing_step(locals())
332                    variant.execute_preprocessing_step(locals())
333                    for (obj, attr), val in variant.values.items():
334                        if not isinstance(obj, str):
335                            obj.SetAttribute(attr, val[step_num])
336                    pfds.initialize_and_run_sim()
337                    sim_res = pfri.export_to_pandas()
338                    sim_res_processed = variant.execute_postprocessing_step(locals())
339                    if sim_res_processed is not None:
340                        sim_res = sim_res_processed
341                    sim_res_processed = self.execute_postprocessing_step(locals())
342                    if sim_res_processed is not None:
343                        sim_res = sim_res_processed
344                    all_sim_res[name] = sim_res
345                    sim_num += 1
346            finally:
347                variant.set_default_values()
348        return all_sim_res

451def format_sig_no_sci(x, sig=3) -> str:
452    """Format figure to three significant digits"""
453    if x == 0:
454        return "0"
455
456    # Determine digits before decimal
457    digits = int(math.floor(math.log10(abs(x)))) + 1
458    decimals = max(sig - digits, 0)
459
460    return f"{x:.{decimals}f}"
