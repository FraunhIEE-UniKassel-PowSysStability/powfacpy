377def create_elm_object(pf_elm_object) -> PFElm:
378    class_name: str = pf_elm_object.GetClassName()
379    elm_class: PFGeneral = elm_classes.get(class_name)
380    if elm_class:
381        return elm_class(pf_elm_object)
382    else:
383        return PFElm(pf_elm_object)
