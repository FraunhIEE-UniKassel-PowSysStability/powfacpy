 87        self,
 88        obj: PFGeneral,
 89        variables: str | list[str],
 90        graphics_page: str | GrpPage | SetVipage = None,
 91        plot: VisPlot | PltLinebarplot | PltVectorplot = None,
 92        **kwargs,
 93    ) -> None:
 94        """Plot varibales. Variables must have been added to the monitored
 95        variables before (e.g. using 'add_results_variable').
 96
 97        Args:
 98            obj (PFGeneral): Object (e.g. of class 'Elm...') of which variables are plotted.
 99
100            variables (str | list[str]): Variable(s) to be plotted.

109              linewidth: double

238    def set_curve_attributes(
239        self, data_series: PltDataseries | None = None, curve_num: int = -1, **kwargs
240    ) -> None:
241        """Set curve attributes in dataseries object of a plot.
242
243        Args:
244            data_series (PltDataseries, optional): data series of plot. Defaults to None (dataseries of active plot is used).
245
246            curve_num (int, optional): Curve number. Defaults to -1 (last curve).
247
248            kwargs:
249              results_obj: result object used (object or path)
250              linestyle: int
251              linewidth: double
252              color: int
253              label: str
254        """
255        if curve_num > 0:
256            curve_num -= 1
257        if not data_series:
258            data_series = self.get_data_series_of_active_plot()
259        if "linestyle" in kwargs:
260            list_curveTableAttr = data_series.GetAttribute("curveTableLineStyle")
261            list_curveTableAttr[curve_num] = kwargs["linestyle"]
262            data_series.SetAttribute("curveTableLineStyle", list_curveTableAttr)
263        if "linewidth" in kwargs:
264            list_curveTableAttr = data_series.GetAttribute("curveTableLineWidth")
265            list_curveTableAttr[curve_num] = kwargs["linewidth"]
266            data_series.SetAttribute("curveTableLineWidth", list_curveTableAttr)
267        else:
268            # The linewidth must be set to the standard value. Otherwise PF uses
269            # the value from the previous data series (this seems to be a PF bug).
270            list_curveTableAttr = data_series.GetAttribute("curveTableLineWidth")
271            list_curveTableAttr[curve_num] = 100
272            data_series.SetAttribute("curveTableLineWidth", list_curveTableAttr)
273        if "color" in kwargs:
274            list_curveTableAttr = data_series.GetAttribute("curveTableColor")
275            list_curveTableAttr[curve_num] = kwargs["color"]
276            data_series.SetAttribute("curveTableColor", list_curveTableAttr)
277        # The label must be handled differently because PF returns an empty list
278        # if there haven't been any labels specified yet for any of the curves.
279        if "label" in kwargs:
280            list_curveTableAttr = data_series.GetAttribute("curveTableLabel")
281            if list_curveTableAttr:
282                list_curveTableAttr[curve_num] = kwargs["label"]
283            else:
284                list_curveTableAttr = [kwargs["label"]]
285            data_series.SetAttribute("curveTableLabel", list_curveTableAttr)
286        if "results_obj" in kwargs:
287            list_curveTableAttr = data_series.GetAttribute("curveTableResultFile")
288            list_curveTableAttr[curve_num] = (
289                self.act_prj._handle_single_pf_object_or_path_input(
290                    kwargs["results_obj"]
291                )
292            )
293            data_series.SetAttribute("curveTableResultFile", list_curveTableAttr)

381    def copy_graphics_board_content(
382        self,
383        source_study_case: str | IntCase,
384        target_study_cases: str | IntCase | list[str] | list[IntCase],
385        obj_to_copy: str = "*",
386        clear_target_graphics_board: bool = False,
387    ) -> None:
388        """Copy the graphics board content of a study case to another study cases.
389
390        Args:
391            source_study_case (str | IntCase): Source case (path or object)
392
393            target_study_cases (str | IntCase | list[str] | list[IntCase]): Target case(s) (path(s) or object(s))
394
395            obj_to_copy (str, optional): name of objects to be copied from graphics board.(e.g. "*.GrpPage" to copy only the plot pages). Defaults to "*".
396
397            clear_target_graphics_board (bool, optional): If true, the graphics boards of the target cases are cleared before pasting the content. Defaults to False.
398        """
399        currently_active_study_case = (
400            self.act_prj.app.GetActiveStudyCase()
401        )  # Method should not change active case
402        source_study_case = self.act_prj._handle_single_pf_object_or_path_input(
403            source_study_case, include_subfolders=False
404        )
405        source_graphics_board = self.act_prj.get_unique_obj(
406            ".SetDesktop", parent_folder=source_study_case, include_subfolders=False
407        )
408        if not isinstance(target_study_cases, (list, tuple)):
409            target_study_cases = [target_study_cases]
410        for target_study_case in target_study_cases:
411            target_study_case = self.act_prj._handle_single_pf_object_or_path_input(
412                target_study_case, include_subfolders=False
413            )
414            if not target_study_case == source_study_case:
415                target_study_case.Deactivate()  # Writing to active graphics board not possible
416                target_graphics_board = self.act_prj.get_unique_obj(
417                    ".SetDesktop",
418                    parent_folder=target_study_case,
419                    include_subfolders=False,
420                )
421                if clear_target_graphics_board:
422                    self.act_prj.delete_obj(
423                        "*",
424                        parent_folder=target_graphics_board,
425                        error_if_non_existent=False,
426                    )
427                self.act_prj.copy_obj(
428                    obj_to_copy,
429                    target_folder=target_graphics_board,
430                    overwrite=True,
431                    parent_folder=source_graphics_board,
432                )
433        if currently_active_study_case:
434            currently_active_study_case.Activate()  # Activate if it was deactivated

553        """Plot data from a pandas DataFrame inside PF using the COMTRADE format as an intermediate step.
554
555        Args:
556            df (pandas.DataFrame): Frame with time index and columns (labels must be single index) for each variable to be plotted.
557            graphics_page (str | GrpPage | SetVipage, optional): target grapics page. Defaults to None (active page is used).
558            plot (VisPlot | PltLinebarplot | PltVectorplot, optional): target plot. Defaults to None (active is used).

564        """

715    def get_curve_table_attributes_referring_to_data_source(
716        self,
717        plot: VisPlot | PltLinebarplot | PltVectorplot = None,
718        adjust_result_file: bool = True,
719        curves: list[int] | None = None,
720    ) -> dict:
721        """Get the curve table attributes referring to the data source
722        of the curves from a plot (i.e. its data series).
723
724        These attributes are: "curveTableResultFile", "curveTableElement",
725          "curveTableVariable"
726
727        Use this method if the data sources of the curves are of interest.
728        If further attributes on visualisation are of interest, see also the
729        methods:
730          - get_curve_table_attributes
731          - get_curve_table_attributes_referring_to_visualization
732
733        Args:
734            plot (VisPlot | PltLinebarplot | PltVectorplot, optional): path or PF object. If None, the active plot is used. Defaults to None.
735
736            adjust_result_file (bool, optional): Defaults to True.
737              - If False, the list in "curveTableResultFile" is used as is
738              - If True, the list is adjusted depending on the plot settings
739                "useIndividualResults". If "useIndividualResults" is True,
740                the result files from the list "curveTableResultFile" are used
741                by default. If an element of this list is empty, the
742                "userSelectedResultFile" is used. Note that there is a bug in
743                PF so that "autoSelectedResultFile" is always empty as described
744                below.
745
746            curves (list[int] | None, optional): If not None, only the attributes of the curves with the given indices are returned.
747
748        Returns:
749            dict:
750              - keys: attribute names
751              - values: lists with the values for each curve
752        """
753        data_series = self.get_data_series_of_plot(plot)
754        attributes = dict.fromkeys(
755            ["curveTableResultFile", "curveTableElement", "curveTableVariable"]
756        )
757        attributes["curveTableElement"] = data_series.GetAttribute("curveTableElement")
758        attributes["curveTableVariable"] = data_series.GetAttribute(
759            "curveTableVariable"
760        )
761        if adjust_result_file:
762            # PF bug:
763            # Even if 'autoSearchResultsFile' is True, 'userSelectedResultFile' and
764            # not 'autoSelectedResultFile' must be used.
765            # Furthermore, 'autoSearchResultFile' must be deactivated and activated once.
766            auto_search_result_file_original_value = data_series.autoSearchResultFile
767            data_series.autoSearchResultFile = 0
768            data_series.autoSearchResultFile = 1
769            data_series.autoSearchResultFile = auto_search_result_file_original_value
770            attributes["curveTableResultFile"] = [
771                data_series.GetAttribute("userSelectedResultFile")
772            ] * len(attributes["curveTableElement"])
773            if data_series.useIndividualResults:
774                curve_table_result_list = data_series.GetAttribute(
775                    "curveTableResultFile"
776                )
777                for idx, result_file in enumerate(curve_table_result_list):
778                    if result_file:
779                        attributes["curveTableResultFile"][idx] = result_file
780        else:
781            attributes["curveTableResultFile"] = data_series.GetAttribute(
782                "curveTableResultFile"
783            )
784        if curves is not None:
785            for key, value in attributes.items():
786                attributes[key] = [value[curve] for curve in curves]
787        return attributes

240        self, data_series: PltDataseries | None = None, curve_num: int = -1, **kwargs
241    ) -> None:
242        """Set curve attributes in dataseries object of a plot.
243
244        Args:
245            data_series (PltDataseries, optional): data series of plot. Defaults to None (dataseries of active plot is used).
246
247            curve_num (int, optional): Curve number. Defaults to -1 (last curve).
248
249            kwargs:
250              results_obj: result object used (object or path)
251              linestyle: int
252              linewidth: double
253              color: int
254              label: str
255        """
256        if curve_num > 0:
257            curve_num -= 1
258        if not data_series:

260        if "linestyle" in kwargs:

270            # the value from the previous data series (this seems to be a PF bug).
271            list_curveTableAttr = data_series.GetAttribute("curveTableLineWidth")

274        if "color" in kwargs:
275            list_curveTableAttr = data_series.GetAttribute("curveTableColor")
276            list_curveTableAttr[curve_num] = kwargs["color"]
277            data_series.SetAttribute("curveTableColor", list_curveTableAttr)
278        # The label must be handled differently because PF returns an empty list
279        # if there haven't been any labels specified yet for any of the curves.
280        if "label" in kwargs:
281            list_curveTableAttr = data_series.GetAttribute("curveTableLabel")
282            if list_curveTableAttr:
283                list_curveTableAttr[curve_num] = kwargs["label"]
284            else:
285                list_curveTableAttr = [kwargs["label"]]
286            data_series.SetAttribute("curveTableLabel", list_curveTableAttr)
287        if "results_obj" in kwargs:
288            list_curveTableAttr = data_series.GetAttribute("curveTableResultFile")
289            list_curveTableAttr[curve_num] = (
290                self.act_prj._handle_single_pf_object_or_path_input(
291                    kwargs["results_obj"]

546    def plot_from_pandas_using_comtrade(
547        self,
548        df: pandas.DataFrame,
549        graphics_page: str | GrpPage | SetVipage = None,
550        plot: VisPlot | PltLinebarplot | PltVectorplot = None,
551        comtrade_file_dir: str | None = None,
552        comtrade_file_name: str | None = None,
553    ) -> str:
554        """Plot data from a pandas DataFrame inside PF using the COMTRADE format as an intermediate step.
555
556        Args:
557            df (pandas.DataFrame): Frame with time index and columns (labels must be single index) for each variable to be plotted.
558            graphics_page (str | GrpPage | SetVipage, optional): target grapics page. Defaults to None (active page is used).
559            plot (VisPlot | PltLinebarplot | PltVectorplot, optional): target plot. Defaults to None (active is used).
560            comtrade_file_dir (str | None, optional): directory to store COMTRADE file. Defaults to None (project dir is used).
561            comtrade_file_name (str | None, optional): Name of COMTRADE file. Defaults to None (name is created from plot name).
562
563        Returns:
564            str: path of the created COMTRADE file (without extension).
565        """
566        if comtrade_file_dir is None:
567            directory = self.act_prj.get_project_directory() + "\\comtrade_files"
568            if not exists(directory):
569                makedirs(directory)
570        if comtrade_file_name is None:
571            comtrade_file_name = (
572                f"{self.active_graphics_page.loc_name}_{self.active_plot.loc_name}_0"
573            )
574        while exists(f"{directory}\\{comtrade_file_name}.cfg"):
575            comtrade_file_name = (
576                f"{comtrade_file_name[:-1]}{str(int(comtrade_file_name[-1]) + 1)}"
577            )
578        recData = [df.iloc[:, n].to_numpy() for n in range(df.shape[1])]
579        cmtr_write = comtradeWriter("testRec", recData)
580        cmtr_write.filename = f"{directory}\\{comtrade_file_name}"
581        cmtr_write.setChannelID(df.columns.to_list())
582        cmtr_write.createFile()
583        # PF only accepts comtrade standard from 1999, so the first line needs to be changed.
584        with open(cmtr_write.filename + ".cfg") as f:
585            lines = f.readlines()
586        lines[0] = "HSBT,,1999\n"
587        with open(cmtr_write.filename + ".cfg", "w") as f:
588            f.writelines(lines)
589        self.plot_from_comtrade(
590            cmtr_write.filename + ".cfg",
591            df.columns.to_list(),
592            graphics_page=graphics_page,
593            plot=plot,
594        )
595        return cmtr_write.filename

597    def plot_from_csv_using_elm_file(
598        self, file_path: str, variable: str, **kwargs
599    ) -> None:
600        """Use an ElmFile object to plot data from csv file.
601
602        It is generally preferrable and more stable to use the COMTRADE format to plot external data.
603
604        The ElmFile objects are stored in a dummy network because the simulation needs to be run to read the data from the csv file and is not just printed to the plot automatically.
605
606        Args:
607            file_path (str): path to csv file
608            variable (str): varaible name in csv file header
609        """
610        # Deactivate currently active networks
611        active_networks = self.act_prj.get_active_networks(
612            error_if_no_network_is_active=False
613        )
614        for network in active_networks:
615            network.Deactivate()
616        # Activate dummy network
617        elmfiles_network = self.create_dummy_network("network_for_elmfiles")
618        elmfiles_network.Activate()
619        # Add new ElmFile
620        elmfile_num = 1
621        while True:
622            existing_elmfile = self.act_prj.get_unique_obj(
623                "elmfile_" + str(elmfile_num),
624                parent_folder=elmfiles_network,
625                error_if_non_existent=False,
626            )
627            if not existing_elmfile:
628                break
629            elmfile_num += 1
630        elmfile = self.act_prj.create_in_folder(
631            "elmfile_" + str(elmfile_num) + ".ElmFile", elmfiles_network, overwrite=True
632        )
633        self.act_prj.clear_elmres_from_objects_with_status_deleted()
634        elmfile.f_name = file_path + ".csv"
635        # Add ElmREs for ElmFiles
636        active_case = self.act_prj.app.GetActiveStudyCase()
637        elmres_for_elmfiles = self.act_prj.create_in_folder(
638            "elmres_for_elmfiles.ElmRes",
639            active_case,
640            overwrite=False,
641            use_existing=True,
642        )
643        # Plot
644        kwargs["results_obj"] = elmres_for_elmfiles
645        self.plot(elmfile, variable, **kwargs)
646        # Simulate
647        pfds = DynamicSimulation(self.app)
648        cominc = self.act_prj.get_from_study_case("ComInc")
649        initial_elmres = self.act_prj.get_attr(cominc, "p_resvar")
650        self.act_prj.set_attr(cominc, {"p_resvar": elmres_for_elmfiles})
651        pfds.initialize_and_run_sim()
652        self.act_prj.set_attr(cominc, {"p_resvar": initial_elmres})
653        # Reactivate the initial networks
654        elmfiles_network.Deactivate()
655        for network in active_networks:
656            network.Activate()

716    def get_curve_table_attributes_referring_to_data_source(
717        self,
718        plot: VisPlot | PltLinebarplot | PltVectorplot = None,
719        adjust_result_file: bool = True,
720        curves: list[int] | None = None,
721    ) -> dict:
722        """Get the curve table attributes referring to the data source
723        of the curves from a plot (i.e. its data series).
724
725        These attributes are: "curveTableResultFile", "curveTableElement",
726          "curveTableVariable"
727
728        Use this method if the data sources of the curves are of interest.
729        If further attributes on visualisation are of interest, see also the
730        methods:
731          - get_curve_table_attributes
732          - get_curve_table_attributes_referring_to_visualization
733
734        Args:
735            plot (VisPlot | PltLinebarplot | PltVectorplot, optional): path or PF object. If None, the active plot is used. Defaults to None.
736
737            adjust_result_file (bool, optional): Defaults to True.
738              - If False, the list in "curveTableResultFile" is used as is
739              - If True, the list is adjusted depending on the plot settings
740                "useIndividualResults". If "useIndividualResults" is True,
741                the result files from the list "curveTableResultFile" are used
742                by default. If an element of this list is empty, the
743                "userSelectedResultFile" is used. Note that there is a bug in
744                PF so that "autoSelectedResultFile" is always empty as described
745                below.
746
747            curves (list[int] | None, optional): If not None, only the attributes of the curves with the given indices are returned.
748
749        Returns:
750            dict:
751              - keys: attribute names
752              - values: lists with the values for each curve
753        """
754        data_series = self.get_data_series_of_plot(plot)
755        attributes = dict.fromkeys(
756            ["curveTableResultFile", "curveTableElement", "curveTableVariable"]
757        )
758        attributes["curveTableElement"] = data_series.GetAttribute("curveTableElement")
759        attributes["curveTableVariable"] = data_series.GetAttribute(
760            "curveTableVariable"
761        )
762        if adjust_result_file:
763            # PF bug:
764            # Even if 'autoSearchResultsFile' is True, 'userSelectedResultFile' and
765            # not 'autoSelectedResultFile' must be used.
766            # Furthermore, 'autoSearchResultFile' must be deactivated and activated once.
767            auto_search_result_file_original_value = data_series.autoSearchResultFile
768            data_series.autoSearchResultFile = 0
769            data_series.autoSearchResultFile = 1
770            data_series.autoSearchResultFile = auto_search_result_file_original_value
771            attributes["curveTableResultFile"] = [
772                data_series.GetAttribute("userSelectedResultFile")
773            ] * len(attributes["curveTableElement"])
774            if data_series.useIndividualResults:
775                curve_table_result_list = data_series.GetAttribute(
776                    "curveTableResultFile"
777                )
778                for idx, result_file in enumerate(curve_table_result_list):
779                    if result_file:
780                        attributes["curveTableResultFile"][idx] = result_file
781        else:
782            attributes["curveTableResultFile"] = data_series.GetAttribute(
783                "curveTableResultFile"
784            )
785        if curves is not None:
786            for key, value in attributes.items():
787                attributes[key] = [value[curve] for curve in curves]
788        return attributes

792        plot: VisPlot | PltLinebarplot | PltVectorplot | None = None,
793        curves: list[int] | None = None,
794    ) -> dict[str, object]:
795        """Get the curve table attributes of a plot  (i.e. its data series)
796        that refer to the visualisation.
797
798        The return value is a dictionary with
799          - keys: attribute names
800          - values: lists with the values for each curve
801
802        Use this method if the attributes relevant for visualization of the curves are of interest.
803        If further attributes on are of interest, see also the
804        methods:
805          - get_curve_table_attributes
806          - get_curve_table_attributes_referring_to_data_source
807
808        Args:
809            plot (VisPlot | PltLinebarplot | PltVectorplot, optional): path or PF object. If None, the active plot is used. Defaults to None.
810
811        Returns:
812            dict[str, object]: dict with attributes and their values.
813        """
814        data_series = self.get_data_series_of_plot(plot)
815        attributes = dict.fromkeys(
816            [
