powfacpy.applications.database

The database interface is helpful to interact with larger numbers of PF objects in the database (getting, setting, storing or comparing attributes). One use case is for example to define a set of standard parameters for a larger model. Using the interface, one can easily get, store (e.g. in a json file) and later reset a set of parameters.

 39    def get_object_attributes(
 40        self,
 41        objs: Iterable[PFGeneral],
 42        class_attributes: dict = None,
 43        keys_are_pf_obj: bool = True,
 44        truncated_path: str = "",
 45        values_are_pf_obj: bool = True,
 46    ) -> dict[str, dict[str, str]]:
 47        """
 48        Get dictionary with attributes of objects.
 49        The keys of the dictionary are PF objects or their path.
 50        The values are dictionaries with keys (attribute names) and values (attribute values).
 51
 52        Args:
 53            objs(Iterable[PFGeneral]): Iterable with PF objects.
 54
 55            class_attributes(dict, optional): dictionary with keys (class names) and values (iterable with attribute names) to control which attributes are relevant for a class. The class names can contain wildcards ("*"). The attributes can contain '.' to reference objects (e.g. to a referenced type object: 'typ_id.Tan'). Defaults to None.
 56
 57            Example:
 58            ```python
 59            {
 60                "ElmTr2": ["typ_id"],
 61                "*": ["loc_name", ],
 62                "ElmVac": ["bus1", "outserv"],
 63            }
 64            ```
 65            -> "loc_name" is relevant for every class
 66
 67            keys_are_pf_obj (str, optional): If True, keys are PF objects, otherwise their paths are used as keys. Defaults to False.
 68
 69            truncated_path(str, optional): If specified, this path is truncated from the keys (object paths). Defaults to "".
 70            Example:
 71            ```
 72            truncated_path = "Network Model\\Network Data"
 73            # then 'Network Model.IntPrjfolder\\Network Data.IntPrjfolder\\test_database_interface\\Grid.ElmNet\\AC Voltage Source.ElmVac'
 74            # becomes (first part truncated):
 75            'test_database_interface\\Grid.ElmNet\\AC Voltage Source.ElmVac"
 76            ```
 77
 78            values_are_pf_obj (bool, optional): If True and if the value of an attribute is a PF object, it will be used as a value in the returned dictionary. Otherwise its path is returned. Defaults to True.
 79
 80        Returns:
 81            dict[str, dict[str, str]]: PF objects' data
 82            Example:
 83            ```
 84            {
 85                "Network Model.IntPrjfolder\\Network Data.IntPrjfolder\\test_database_interface\\Grid.ElmNet\\AC Voltage Source.ElmVac": {
 86                        "loc_name": "AC Voltage Source",
 87                        "bus1": "Network Model.IntPrjfolder\\Network Data.IntPrjfolder\\test_database_interface\\Grid.ElmNet\\Terminal HV 1.ElmTerm\\Cub_1.StaCubic",
 88                        "outserv": 0
 89                },
 90                "Network Model.IntPrjfolder\\Network Data.IntPrjfolder\\test_database_interface\\Grid.ElmNet\\Terminal HV 1.ElmTerm\\Cub_1.StaCubic": {
 91                        "loc_name": "Cub_1"
 92                },
 93            }
 94            ```
 95        """
 96        if not class_attributes:
 97            class_attributes = {
 98                "*": []
 99            }  # no relevant attributes -> get only keys (object paths)
100        if truncated_path:  # include class names in truncated_path
101            truncated_path = self.act_prj.get_path_of_obj_with_class_names(
102                truncated_path
103            )
104
105        obj_attr_dict = {}
106        for obj in objs:
107            if not keys_are_pf_obj:
108                key = self.act_prj.get_path_of_obj_with_class_names(obj)
109                if truncated_path:
110                    key = PFStringManipulation.truncate_until(
111                        key, truncated_path + "\\"
112                    )
113            else:
114                key = obj
115            obj_attr_dict[key] = {}
116            for class_name in class_attributes.keys():
117                if fnmatchcase(
118                    obj.GetClassName(), class_name
119                ):  # fnmatch allows to use wildcards ("*")
120                    for attr in class_attributes[class_name]:
121                        val = self._handle_attribute_type_for_reading(
122                            obj, attr, not values_are_pf_obj
123                        )
124                        if not (val is False):
125                            obj_attr_dict[key][attr] = val
126        return obj_attr_dict

128    def set_pf_obj_attribute_values(
129        self, obj_attr_dict: dict, added_path: None | str = None
130    ) -> None:
131        """
132        Writes the data of a dict to the PF database
133        (the dict can be created e.g. with 'get_object_attributes').
134        There are two options for the format of the dict:
135        - keys: PF object or path, values: dict[attr, value]
136        - keys: tuple(PF object or path, attr ), values: value
137
138        Arguments:
139          obj_attr_dict (dict): data
140          added_path (str): Assumes that the objects' path is relative to a parent folder inside the project. Adds 'added_path' to the paths .
141        """
142        if not isinstance(list(obj_attr_dict)[0], tuple):
143            for obj, attr_key_val in obj_attr_dict.items():
144                obj = self._handle_obj_type(obj, added_path=added_path)
145                for attr, value in attr_key_val.items():
146                    obj, attr = self._handle_attribute_type_for_writing(obj, attr)
147                    value = self._handle_value_type(obj, attr, value)
148                    self.act_prj.set_attr(obj, {attr: value})
149        else:
150            for (obj, attr), value in obj_attr_dict.items():
151                obj = self._handle_obj_type(obj, added_path=added_path)
152                obj, attr = self._handle_attribute_type_for_writing(obj, attr)
153                value = self._handle_value_type(obj, attr, value)
154                self.act_prj.set_attr(obj, {attr: value})

156    def _handle_obj_type(
157        self, obj: PFGeneral | str, added_path: None | str = None
158    ) -> PFGeneral:
159        """Converts obj specified by their path to the actual PF object. 
160
161        Args:
162            obj (PFGeneral | str): _description_
163            added_path (None | str, optional): Added path prefix. Defaults to None.
164
165        Returns:
166            PFGeneral: object
167        """ """"""
168        if isinstance(obj, str):
169            if added_path is not None:
170                obj = added_path + "\\" + obj
171            obj = self.act_prj.get_unique_obj(obj, include_subfolders=False)
172        return obj

174    def _handle_value_type(self, obj: PFGeneral, attr: str, value: Any):
175        """Converts path strings in attribute values to PF objects.
176
177        Args:
178            obj (PFGeneral): PF object
179            attr (str): attribute
180            value (Any): value of attribute

182        Returns:
183            PF object | str:
184        """
185        if isinstance(value, str) and not isinstance(
186            obj.GetAttributeType(attr), str
187        ):  # Then the true attribute type is a PF object
188            return self.act_prj.get_unique_obj(value)
189        else:

192    def _handle_attribute_type_for_reading(
193        self,
194        obj: PFGeneral,
195        attr: str,
196        paths_are_used_for_obj: bool = False,
197        truncated_path: str = "",
198    ) -> PFGeneral | str:
199        """
200        Various ways to handle attributes which are PF objects (i.e. a PF object holds another PF object as an attribute, e.g. a 'type') when
201        reading from the PF database.
202
203        PF objects can be read as
204          - a PF object (pf_obj_handling="original_pf_obj")
205          - the path of the object (pf_obj_handling="path", default)
206
207        Arguments:
208            obj (PFGeneral): PF object
209            attr (str): Attribute of the object
210            paths_are_used_for_obj (bool, optional): If True, the path of the object is returned instead of the object itself. Defaults to False.
211            truncated_path: If 'paths_are_used_for_obj' is True, then 'truncated_path' is truncated from the beginning of the object's path. Defaults to "".
212
213        Returns:
214            PFGeneral | str: PF object or its path
215        """
216        attr = attr.split(".")
217        attr_value = obj
218        for a in attr:
219            if attr_value is None:
220                return False
221            attr_value = self.act_prj.get_attr(attr_value, a)
222        if isinstance(attr_value, (str, int, float)) or not attr_value:
223            return attr_value
224        else:  # Then it must be a PF object
225            if not paths_are_used_for_obj:
226                return attr_value
227            else:
228                if truncated_path:
229                    return self.act_prj.get_path_of_obj_with_class_names(truncated_path)
230                else:
231                    return self.act_prj.get_path_of_obj_with_class_names(attr_value)
232
233    def _handle_attribute_type_for_writing(
234        self, obj: PFGeneral, attr: str
235    ) -> tuple[PFGeneral, str]:
236        """If the attr contains '.' (reference to another obj), return the referenced object and its attribute name. Otherwise just return the input arguments
237
238        Args:
239            obj (PFGeneral): PF object
240            attr (str): attribute
241
242        Returns:
243            tuple[PFGeneral, str]: _description_
244        """
245        if "." in attr:
246            attr = attr.split(".")
247            for a in attr[0:-1]:
248                obj = self.act_prj.get_attr(obj, a)
249            attr = attr[-1]
250        return obj, attr

252    def make_loc_name_unique(
253        self,
254        pf_classes: list = ["Elm*"],
255        objs: Iterable[PFGeneral] | None = None,
256        suffix_separator: str = "_",
257    ) -> dict[str, int]:
258        """Change duplicate loc_name of calculation relevant objects of the same class to ensure that they are unique. Number the loc_name with a suffix '_n' where n is the count. Useful for example when loc_name are used as keys in a dict or labels in DataFrames (e.g. when exporting to pandapower format).
259
260        Args:
261            pf_classes (Iterable[str] | str, optional): Classes to be considered. Defaults to ["Elm*"].
262            objs (Iterable[PFGeneral] | None): Selection ob objects to be considered. Defaults to None (all calculation relevant).
263            suffix_separator (str): Suffix used to change duplicate names. Defaults to "_".
264
265        Returns:
266            dict[str, int]:
267                keys: loc_names of objects;
268                values: number of occurrences of loc_name in the original dataset.
269        """
270
271        def count_and_rename_objs() -> None:
272            for obj in objs:
273                name_with_class = obj.loc_name + "." + obj.GetClassName()
274                if not name_with_class in obj_count.keys():
275                    obj_count[name_with_class] = 1
276                else:
277                    obj_count[name_with_class] += 1
278                    obj.loc_name = (
279                        obj.loc_name + f"{suffix_separator}{obj_count[name_with_class]}"
280                    )
281
282        obj_count = {}
283        if objs is not None:
284            count_and_rename_objs()
285        else:
286            all_pf_classes = []
287            for pf_class in pf_classes:
288                all_pf_classes += self.get_class_names(pf_class)
289            objs = []
290            for class_name in all_pf_classes:
291                objs += self.act_prj.app.GetCalcRelevantObjects(f"*.{class_name}")
292            count_and_rename_objs()
293        return obj_count

295    def get_class_names(self, name: str = "*") -> list[str]:
296        """Get match cases of 'name' with all PF classes from powfacpy.pf_classes.protocols."""
297        class_names = []
298        for cls_name, _cls in inspect.getmembers(
299            importlib.import_module("powfacpy.pf_classes.protocols"), inspect.isclass
300        ):
301            if fnmatchcase(cls_name, name):
302                class_names.append(cls_name)
303        return class_names

332    def obj_to_str(
333        self, truncate: None | str = None, inplace: bool = False
334    ) -> DatabaseDict:
335        """Transform PF objects in keys to path strings (PF objects in values of attributes are not affected).
336
337        Args:
338            truncate (None | str, optional): Truncate prefix from resulting path. Defaults to None.
339            inplace (bool, optional): If True, this dict is modified. Defaults to False (return modified dict).
340
341        Returns:
342            DatabaseDict: modified dict
343        """
344        d_new = {}
345        for obj, val in self.items():
346            if not isinstance(obj, str):
347                obj = self.act_prj.get_path_of_object_in_active_project(obj)
348            if truncate is not None:
349                obj = obj.removeprefix(truncate)
350            d_new[obj] = val
351        if inplace:
352            self.reset(d_new)
353        return DatabaseDict(d_new)
354
355    def keys_to_obj_attr_str(
356        self, truncate: None | str = None, inplace: bool = False
357    ) -> DatabaseDict:
358        """Transform PF objects in keys to strings containing object paths and attributes.
359
360        Args:
361            truncate (None | str, optional): Truncate prefix from resulting path. Defaults to None.
362            inplace (bool, optional): If True, this dict is modified. Defaults to False (return modified dict).
363
364        Returns:
365            DatabaseDict: modified dict
366        """
367        d_strings = self.obj_to_str(truncate=truncate)
368        d_new = {}
369        for obj, dict_attr_val in d_strings.items():
370            for attr, val in dict_attr_val.items():
371                d_new[obj + "\\" + attr] = val
372        if inplace:
373            self.reset(d_new)
374        return DatabaseDict(self.app, d_new)

376    def get_obj_attribute_strings(self, truncate: None | str = None) -> list[str]:
377        """Get list of object attribute strings.

379        Args:

381
382        Returns:
383            list[str]: object attribute strings
384        """
385        dbd = self.keys_to_obj_attr_str(truncate=truncate)
386        return list(dbd.d.keys())

398        """Set attribute values of PF objects in dict using 'values'.
399
400        Args:

402        """
403        n = 0
404        for obj, dict_attr_val in self.items():
405            for attr in dict_attr_val.keys():
406                self[obj][attr] = values[n]

409        """Set values of DatabaseDict in PF objects."""
410        for obj, dict_attr_val in self.items():

415        """Set values of DatabaseDict in PF objects and store the original values in this dict."""
416        for obj, dict_attr_val in self.items():
417            for attr, val in dict_attr_val.items():
418                old_value = obj.GetAttribute(attr)
