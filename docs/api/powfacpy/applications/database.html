powfacpy.applications.database

The database interface is helpful to interact with larger numbers of PF objects in the database (getting, setting, storing or comparing attributes). One use case is for example to define a set of standard parameters for a larger model. Using the interface, one can easily get, store (e.g. in a json file) and later reset a set of parameters.

 41    def get_object_attributes(
 42        self,
 43        objs: Iterable[PFGeneral],
 44        class_attributes: dict = None,
 45        keys_are_pf_obj: bool = True,
 46        truncated_path: str = "",
 47        values_are_pf_obj: bool = True,
 48    ) -> dict[str, dict[str, str]]:
 49        """
 50        Get dictionary with attributes of objects.
 51        The keys of the dictionary are PF objects or their path.
 52        The values are dictionaries with keys (attribute names) and values (attribute values).
 53
 54        Args:
 55            objs(Iterable[PFGeneral]): Iterable with PF objects.
 56
 57            class_attributes(dict, optional): dictionary with keys (class names) and values (iterable with attribute names) to control which attributes are relevant for a class. The class names can contain wildcards ("*"). The attributes can contain '.' to reference objects (e.g. to a referenced type object: 'typ_id.Tan'). Defaults to None.
 58
 59            Example:
 60            ```python
 61            {
 62                "ElmTr2": ["typ_id"],
 63                "*": ["loc_name", ],
 64                "ElmVac": ["bus1", "outserv"],
 65            }
 66            ```
 67            -> "loc_name" is relevant for every class
 68
 69            keys_are_pf_obj (str, optional): If True, keys are PF objects, otherwise their paths are used as keys. Defaults to False.
 70
 71            truncated_path(str, optional): If specified, this path is truncated from the keys (object paths). Defaults to "".
 72            Example:
 73            ```
 74            truncated_path = "Network Model\\Network Data"
 75            # then 'Network Model.IntPrjfolder\\Network Data.IntPrjfolder\\test_database_interface\\Grid.ElmNet\\AC Voltage Source.ElmVac'
 76            # becomes (first part truncated):
 77            'test_database_interface\\Grid.ElmNet\\AC Voltage Source.ElmVac"
 78            ```
 79
 80            values_are_pf_obj (bool, optional): If True and if the value of an attribute is a PF object, it will be used as a value in the returned dictionary. Otherwise its path is returned. Defaults to True.
 81
 82        Returns:
 83            dict[str, dict[str, str]]: PF objects' data
 84            Example:
 85            ```
 86            {
 87                "Network Model.IntPrjfolder\\Network Data.IntPrjfolder\\test_database_interface\\Grid.ElmNet\\AC Voltage Source.ElmVac": {
 88                        "loc_name": "AC Voltage Source",
 89                        "bus1": "Network Model.IntPrjfolder\\Network Data.IntPrjfolder\\test_database_interface\\Grid.ElmNet\\Terminal HV 1.ElmTerm\\Cub_1.StaCubic",
 90                        "outserv": 0
 91                },
 92                "Network Model.IntPrjfolder\\Network Data.IntPrjfolder\\test_database_interface\\Grid.ElmNet\\Terminal HV 1.ElmTerm\\Cub_1.StaCubic": {
 93                        "loc_name": "Cub_1"
 94                },
 95            }
 96            ```
 97        """
 98        if class_attributes is None:
 99            class_attributes = {
100                "*": []
101            }  # no relevant attributes -> get only keys (object paths)
102        if truncated_path:  # include class names in truncated_path
103            truncated_path = self.act_prj.get_path_of_obj_with_class_names(
104                truncated_path
105            )
106
107        obj_attr_dict = {}
108        for obj in objs:
109            if not keys_are_pf_obj:
110                key = self.act_prj.get_path_of_obj_with_class_names(obj)
111                if truncated_path:
112                    key = PFStringManipulation.truncate_until(
113                        key, truncated_path + "\\"
114                    )
115            else:
116                key = obj
117            obj_attr_dict[key] = {}
118            for class_name in class_attributes.keys():
119                if fnmatchcase(
120                    obj.GetClassName(), class_name
121                ):  # fnmatch allows to use wildcards ("*")
122                    for attr in class_attributes[class_name]:
123                        val = self._handle_attribute_type_for_reading(
124                            obj, attr, not values_are_pf_obj
125                        )
126                        if not (val is False):
127                            obj_attr_dict[key][attr] = val
128        return obj_attr_dict

130    def set_pf_obj_attribute_values(
131        self, obj_attr_dict: dict, added_path: None | str = None
132    ) -> None:
133        """
134        Writes the data of a dict to the PF database
135        (the dict can be created e.g. with 'get_object_attributes').
136        There are two options for the format of the dict (corresponding to the output of 'get_object_attributes' or 'get_obj_attribute_strings'):
137        - keys: PF object or path, values: dict[attr, value]
138        - keys: tuple(PF object or path, attr ), values: value
139
140        Arguments:
141          obj_attr_dict (dict): data
142          added_path (str): Assumes that the objects' path is relative to a parent folder inside the project. Adds 'added_path' in order to be relative to the project root.
143        """
144        if not isinstance(list(obj_attr_dict)[0], tuple):
145            for obj, attr_key_val in obj_attr_dict.items():
146                obj = self._handle_obj_type(obj, added_path=added_path)
147                for attr, value in attr_key_val.items():
148                    obj, attr = self._handle_attribute_type_for_writing(obj, attr)
149                    value = self._handle_value_type(obj, attr, value)
150                    self.act_prj.set_attr(obj, {attr: value})
151        else:
152            for (obj, attr), value in obj_attr_dict.items():
153                obj = self._handle_obj_type(obj, added_path=added_path)
154                obj, attr = self._handle_attribute_type_for_writing(obj, attr)
155                value = self._handle_value_type(obj, attr, value)
156                self.act_prj.set_attr(obj, {attr: value})

194    def _handle_attribute_type_for_reading(
195        self,
196        obj: PFGeneral,
197        attr: str,
198        paths_are_used_for_obj: bool = False,
199        truncated_path: str = "",
200    ) -> PFGeneral | str:
201        """
202        Various ways to handle attributes which are PF objects (i.e. a PF object holds another PF object as an attribute, e.g. a 'type') when
203        reading from the PF database.
204
205        PF objects can be read as
206          - a PF object (pf_obj_handling="original_pf_obj")
207          - the path of the object (pf_obj_handling="path", default)
208
209        Arguments:
210            obj (PFGeneral): PF object
211            attr (str): Attribute of the object
212            paths_are_used_for_obj (bool, optional): If True, the path of the object is returned instead of the object itself. Defaults to False.
213            truncated_path: If 'paths_are_used_for_obj' is True, then 'truncated_path' is truncated from the beginning of the object's path. Defaults to "".
214
215        Returns:
216            PFGeneral | str: PF object or its path
217        """
218        attr = attr.split(".")
219        attr_value = obj
220        for a in attr:
221            if attr_value is None:
222                return False
223            attr_value = self.act_prj.get_attr(attr_value, a)
224        if isinstance(attr_value, (str, int, float)) or not attr_value:
225            return attr_value
226        else:  # Then it must be a PF object
227            if not paths_are_used_for_obj:
228                return attr_value
229            else:
230                if truncated_path:
231                    return self.act_prj.get_path_of_obj_with_class_names(truncated_path)
232                else:
233                    return self.act_prj.get_path_of_obj_with_class_names(attr_value)

235    def _handle_attribute_type_for_writing(
236        self, obj: PFGeneral, attr: str
237    ) -> tuple[PFGeneral, str]:
238        """If the attr contains '.' (reference to another obj), return the referenced object and its attribute name. Otherwise just return the input arguments
239
240        Args:
241            obj (PFGeneral): PF object
242            attr (str): attribute
243
244        Returns:
245            tuple[PFGeneral, str]: _description_
246        """
247        if "." in attr:
248            attr = attr.split(".")
249            for a in attr[0:-1]:
250                obj = self.act_prj.get_attr(obj, a)
251            attr = attr[-1]
252        return obj, attr

254    def make_loc_name_unique(
255        self,
256        pf_classes: list = ["Elm*"],
257        objs: Iterable[PFGeneral] | None = None,
258        suffix_separator: str = "_",
259    ) -> dict[str, int]:
260        """Change duplicate loc_name of calculation relevant objects of the same class to ensure that they are unique. Number the loc_name with a suffix '_n' where n is the count. Useful for example when loc_name are used as keys in a dict or labels in DataFrames (e.g. when exporting to pandapower format).
261
262        Args:
263            pf_classes (Iterable[str] | str, optional): Classes to be considered. Defaults to ["Elm*"].
264            objs (Iterable[PFGeneral] | None): Selection ob objects to be considered. Defaults to None (all calculation relevant).
265            suffix_separator (str): Suffix used to change duplicate names. Defaults to "_".
266
267        Returns:
268            dict[str, int]:
269                keys: loc_names of objects;
270                values: number of occurrences of loc_name in the original dataset.
271        """
272
273        def count_and_rename_objs() -> None:
274            for obj in objs:
275                name_with_class = obj.loc_name + "." + obj.GetClassName()
276                if not name_with_class in obj_count.keys():
277                    obj_count[name_with_class] = 1
278                else:
279                    obj_count[name_with_class] += 1
280                    obj.loc_name = (
281                        obj.loc_name + f"{suffix_separator}{obj_count[name_with_class]}"
282                    )
283
284        obj_count = {}
285        if objs is not None:
286            count_and_rename_objs()
287        else:
288            all_pf_classes = []
289            for pf_class in pf_classes:
290                all_pf_classes += self.get_class_names(pf_class)
291            objs = []
292            for class_name in all_pf_classes:
293                objs += self.act_prj.app.GetCalcRelevantObjects(f"*.{class_name}")
294            count_and_rename_objs()
295        return obj_count

307    def get_result_variables(
308        self,
309        class_name: str,
310        vars: list | str,
311        simulation_type: str = "RMS_Bal",
312        objs: list | None = None,
313        index_format: str = "name",
314    ) -> pd.DataFrame:
315        """Get result variables for a specified class and simulation type.
316
317        Args:
318            class_name (str): Name of the class (e.g. "ElmSym", "ElmTr2", "ElmGenstat", ...)
319            vars (list | str): list of variable names or a string pattern (e.g. "e_*") to filter the variable names of the specified class.
320            simulation_type (str, optional): "Basic", "LF_Bal", "LF_Unbal", "RMS_Bal", "RMS_Unbal", "EMT", "Sensitivities_Bal". Defaults to "RMS_Bal".
321            objs (list | None, optional): List of objects (class according to 'class_name) to get result variables for. If None, all calculation relevant objects of the specified class are used. Defaults to None.
322            index_format (str, optional): Format of the index of the returned DataFrame. Options: "name" (loc_name, default), "path", "obj". Defaults to "name".
323
324        Returns:
325            pd.DataFrame: result variables of the specified class and simulation type. The index is the loc_name of the objects, the columns are a MultiIndex with the variable name and its description.
326        """
327        enum_class = getattr(getattr(ResVar, simulation_type), class_name)
328        if isinstance(vars, str):
329            vars = vars.replace(":", "_")
330            vars = filter(enum_class.__members__.keys(), vars)
331        if objs is None:
332            objs = self.act_prj.get_calc_relevant_obj("*." + class_name)  
333        var_values = {var: [None]*len(objs) for var in vars}
334        for var in vars:
335            for m, obj in enumerate(objs):
336                attr = enum_class[var].value
337                if obj.HasAttribute(attr):
338                    val = obj.GetAttribute(attr)
339                    if not val:
340                        val = None
341                else:
342                    val = None        
343                var_values[var][m] = val
344        df = pd.DataFrame(var_values)   
345        if index_format == "path":
346            df.index = [self.act_prj.get_path_of_object_in_active_project(o) for o in objs]
347        elif index_format == "obj":
348            df.index = objs
349        else:
350            df.index = [o.loc_name for o in objs]
351        df.columns = pd.MultiIndex.from_tuples(
352            [(enum_class[var].value, enum_class[var].__doc__) 
353                for var in vars]
354        )
355        return df
