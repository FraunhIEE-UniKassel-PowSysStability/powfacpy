 72    def get_eigenvalues_of_current_state(
 73        self, commod_parameters: dict[str, object] = {}
 74    ) -> pd.DataFrame:
 75        """Get the eigenvalues of the current state of the system.
 76
 77        Uses the modal analysis command (ComMod) to calculate the eigenvalues (Eigenvectors and participation factors are omitted). The operating point of the current simulation time is used.
 78        Then uses result export (ComRes) to export the eigenvalues to csv, which is then read to a pandas DataFrame.
 79
 80        Args:
 81            commod_parameters (dict[str, str], optional): Additional parameter settings of modal analysis command (ComMod). Defaults to {}.
 82
 83        Returns:
 84            pd.DataFrame: pandas DataFrame with columns "real in 1/s", "imag in rad/s"
 85        """
 86        commod: ComMod = self.act_prj.get_from_study_case("ComMod")
 87        commod.iLeft = False
 88        commod.iRight = False
 89        commod.iPart = False
 90        commod.initMode = 0
 91        if commod_parameters:
 92            self.act_prj.set_attr(commod, commod_parameters)
 93        commod.Execute()
 94
 95        elmres: ElmRes = commod.ResultFile
 96        comres: ComRes = self.act_prj.app.GetFromStudyCase("ComRes")
 97        comres.pResult = elmres
 98        comres.iopt_csel = 0  # export all variables
 99        comres.f_name = getcwd() + "\\temp.csv"
100        comres.iopt_exp = 6  # export to csv format
101        try:
102            comres.Execute()
103            df = pd.read_csv(
104                comres.f_name, encoding="ISO-8859-1", header=[0, 1], index_col=0
105            )
106        finally:
107            remove(comres.f_name)
108        df.columns = ["real in 1/s", "imag in rad/s"]
109        return df

185    def get_parameters_of_dsl_models_in_composite_model(
186        self,
187        composite_model: ElmComp,
188        single_dict_for_all_dsl_models: bool = False,
189    ):
190        """
191        Returns a dictionary with the parameter names (of the block definition)
192        and values of all dsl models inside a composite model.
193
194        dsl lookup variables (e.g. 'array_*', 'omatrix_*',.. ) are ignored.
195
196        TODO: this method should be in the CompositeModel class
197
198        Arguments:
199          composite_model: ElmComp or its path
200
201          single_dict_for_all_dsl_models:
202            - If true, a single dictionary with the parameters of all dsl models is returned (no distinction is made between the dsl models).
203            This assumes that a parameter that occurs in several dsl
204            models has the same value.
205            Example: {"a": 1, "b":0, "c":2}
206            - If false, the returned dictionary contains dictionaries for each dsl model.
207            Example:
208              {
209                "controller_a": {"a": 1, "b":0}
210                "controller_b": {"a": 5, "c":2}
211              }
212        """
213        composite_model = self.act_prj._handle_single_pf_object_or_path_input(
214            composite_model
215        )
216        composite_model.SlotUpdate()
217        dsl_models = self.get_dsl_models_inside_composite_model(composite_model)
218        all_models_params_dict = {}
219        for dsl_model in dsl_models:
220            parameter_names_incl_dsl_lookup = self.get_dsl_model_parameter_names(
221                dsl_model
222            )
223            # Ignore dsl lookup arrays/matrices
224            parameter_names = []
225            if parameter_names_incl_dsl_lookup:
226                for parameter_name in parameter_names_incl_dsl_lookup:
227                    if not self.is_dsl_lookup_arrays_and_matrices_name(parameter_name):
228                        parameter_names.append(parameter_name)
229
230            if parameter_names:
231                parameter_names_and_values = self.act_prj.get_attr(
232                    dsl_model, parameter_names
233                )
234                if single_dict_for_all_dsl_models:
235                    for param_name in parameter_names:
236                        if param_name in all_models_params_dict:
237                            if (
238                                all_models_params_dict[param_name]
239                                != parameter_names_and_values[param_name]
240                            ):
241                                raise PFInconsistentParamValueOfDSLModelInCompositeModel(
242                                    param_name, composite_model
243                                )
244                    all_models_params_dict = {
245                        **all_models_params_dict,
246                        **parameter_names_and_values,
247                    }
248                else:
249                    all_models_params_dict[dsl_model.loc_name] = (
250                        parameter_names_and_values
251                    )
252            elif not single_dict_for_all_dsl_models:
253                all_models_params_dict[dsl_model.loc_name] = {}
254        return all_models_params_dict

256    def set_parameters_of_dsl_models_in_composite_model(
257        self,
258        composite_model,
259        models_params_dict,
260        single_dict_for_all_dsl_models=False,
261    ) -> None:
262        """
263        Set the parameters of the dsl models (i.e. of its block definition) in
264        a composite model.
265
266        Args:
267          composite_model: ElmComp or its path
268          models_params_dict: dictionary with model parameters and values
269          single_dict_for_all_dsl_models:
270            - If true, models_params_dict is a single dictionary that is used to set the parameters of all dsl models.
271            Example: {"a": 1, "b":0, "c":2} -> if a dsl model has an attribute ("a","b",
272            "c"), the value is set, otherwise it is ignored.
273            - If false, models_params_dict contains dictionaries for each dsl model.
274            Example:
275              {
276              "controller_a": {"a": 1, "b":0}
277              "controller_b": {"a": 5, "c":2}
278              }
279        """
280        composite_model = self.act_prj._handle_single_pf_object_or_path_input(
281            composite_model
282        )
283        if not single_dict_for_all_dsl_models:
284            for dsl_model, parameter_value_dict in models_params_dict.items():
285                dsl_model = self.act_prj.get_unique_obj(
286                    dsl_model + ".ElmDsl",
287                    parent_folder=composite_model,
288                    include_subfolders=True,
289                )
290                self.act_prj.set_attr(dsl_model, parameter_value_dict)
291        else:
292            dsl_models = self.get_dsl_models_inside_composite_model(composite_model)
293            for dsl_model in dsl_models:
294                for param_name, value in models_params_dict.items():
295                    if dsl_model.HasAttribute(param_name):
296                        dsl_model.SetAttribute(param_name, value)

300        """

316    @staticmethod
317    def set_dsl_obj_array(
318        dsl_obj, rows: list, array_num: int | None = None, size_included_in_array=True
319    ) -> None:
320        """Set the array of a DSL object ('Advanced 1' tab).
321        The array_num specifies which array is set (if None,
322        all arrays/colums are set).
323        If size_included_in_array=True, the first row (where the size
324        of the array is specified) is included.
325        """
326        if not size_included_in_array:
327            if not array_num:
328                dsl_obj.SetAttribute("matrix:0", [len(rows)] * len(rows[0]))
329            else:
330                complete_row = dsl_obj.GetAttribute("matrix:0")
331                complete_row[(array_num - 1) * 2] = len(rows)
332                dsl_obj.SetAttribute("matrix:0", complete_row)
333        for row_num, row in enumerate(rows):
334            if not size_included_in_array:
335                attrib = "matrix:" + str(row_num + 1)
336            else:
337                attrib = "matrix:" + str(row_num)
338            if array_num is None:
339                dsl_obj.SetAttribute(attrib, row)
340            else:
341                complete_row = dsl_obj.GetAttribute(attrib)
342                complete_row[(array_num - 1) * 2] = row[0]
343                complete_row[(array_num - 1) * 2 + 1] = row[1]
344                dsl_obj.SetAttribute(attrib, list(complete_row))

346    @staticmethod
347    def set_dsl_obj_array_from_pandas_series(
348        dsl_obj, series: pd.Series, array_num: int | None = None
349    ) -> None:
350        rows = [
351            [float(time), float(value)]
352            for time, value in zip(series.index.values, series.values)
353        ]
354        DynamicSimulation.set_dsl_obj_array(
355            dsl_obj, rows, array_num=array_num, size_included_in_array=False
356        )

358    @staticmethod
359    def get_dsl_obj_array(dsl_obj, array_num=None, size_included_in_array=True):
360        """Get the array of DSL object ('Advanced 1' tab).
361        The array_num specifies which array is returend (if None,
362        all arays/colmns are returned).
363        If size_included_in_array=True, the first row (where the size
364        of the array is spedified) is included.
365        """
366        if not array_num:
367            number_of_rows = int(max(dsl_obj.GetAttribute("matrix:0"))) + 1
368        else:
369            number_of_rows = (
370                int(dsl_obj.GetAttribute("matrix:0")[(array_num - 1) * 2]) + 1
371            )
372        array = []
373        for row_num in range(number_of_rows):
374            if row_num == 0 and not size_included_in_array:
375                continue
376            attrib = "matrix:" + str(row_num)
377            row = dsl_obj.GetAttribute(attrib)
378            if array_num:
379                row = [row[(array_num - 1) * 2], row[(array_num - 1) * 2 + 1]]
380            array.append(row)
381        return array

370            number_of_rows = (
371                int(dsl_obj.GetAttribute("matrix:0")[(array_num - 1) * 2]) + 1
372            )
373        array = []
374        for row_num in range(number_of_rows):
375            if row_num == 0 and not size_included_in_array:

382        return array
