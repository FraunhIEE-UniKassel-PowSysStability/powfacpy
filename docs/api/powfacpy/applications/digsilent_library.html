powfacpy.applications.digsilent_library

Navigate the global DIgSILENT library with the folder names shown in the GUI.

PowerFactory's built-in ("system") library folders have coded loc_names - Templ, TemplGfc, TemplPv - while the GUI (and the documentation) shows translated names like Templates / Grid-forming Converters. The Python API only accepts the coded names, so a GUI-style path such as r"Templates\Grid-forming Converters" cannot be resolved directly.

DigsilentLibrary bridges the two using PowerFactory's own localisation database (<install>/localisation/en-GB/data.db), so no hand-maintained mapping is needed and user-created subfolders (whose loc_name already is the displayed name) just work.

 74    def localisation_db_path(self) -> Path:
 75        """Path of PowerFactory's English localisation database."""
 76        return (
 77            Path(self.app.GetInstallationDirectory())
 78            / "localisation"
 79            / "en-GB"
 80            / "data.db"

 83    @cached_property
 84    def folder_name_translations(self) -> dict[str, str]:
 85        """`{loc_name: displayed_name}` for the built-in (system) library folders.
 86
 87        Read from `localisation_db_path`; on any failure, falls back to
 88        `_FALLBACK_FOLDER_NAMES` and warns.
 89        """
 90        path = self.localisation_db_path
 91        try:
 92            with sqlite3.connect(f"file:{path}?mode=ro", uri=True) as con:
 93                rows = con.execute(
 94                    "SELECT param, message FROM Message "
 95                    "WHERE class = 'IntFolder' AND param LIKE 'skey%'"
 96                ).fetchall()
 97            return {param[len("skey") :]: message for param, message in rows}
 98        except (sqlite3.Error, OSError) as error:
 99            warn(
100                f"Could not read the PowerFactory localisation database at '{path}' "
101                f"({error}); falling back to a built-in name table (Templates branch only).",
102                RuntimeWarning,
103            )
104            return dict(_FALLBACK_FOLDER_NAMES)

123    def get_object(
124        self, display_path: str, error_if_non_existent: bool = True
125    ) -> PFGeneral | None:
126        """Resolve a GUI-style path under the global library.
127
128        Each path segment is matched against the child objects' displayed names
129        (translated `loc_name`) and, as a fallback, their literal `loc_name`.
130        `"\\"` and `"/"` are both accepted as separators.
131
132        Args:
133            display_path: e.g. `r"Templates\\Grid-forming Converters"`. Empty
134                string / `"\\"` returns the library root.
135            error_if_non_existent: raise if a segment cannot be resolved;
136                otherwise return None.
137        """
138        node: PFGeneral = self.root
139        for segment in [s for s in _re_split(r"[\\/]", display_path) if s]:
140            match = self._child_by_name(node, segment)
141            if match is None:
142                if not error_if_non_existent:
143                    return None
144                raise PFInterfaceError(
145                    f"'{segment}' not found under '{self.gui_path(node)}'. "
146                    f"Available: {sorted(self.list_contents(self.gui_path(node)))}."
147                )
148            node = match
149        return node

168    def list_contents(
169        self, display_path: str = "", classes: tuple[str, ...] | None = None
170    ) -> dict[str, PFGeneral]:
171        """`{displayed_name: object}` of the children of the folder at `display_path`.
172
173        Args:
174            display_path: folder to list (default: the library root).
175            classes: keep only these PF class names (default: all).
176        """
177        parent = self.get_object(display_path)
178        result: dict[str, PFGeneral] = {}
179        for child in parent.GetContents():
180            if classes and child.GetClassName() not in classes:
181                continue
182            result[self.gui_name(child)] = child

189    def gui_path(self, obj: PFGeneral) -> str:
190        """GUI-style path of a library object, relative to the library root."""
191        library_full_name = self.root.GetFullName()
192        full_name = obj.GetFullName()
193        if not full_name.startswith(library_full_name):
194            return full_name
195        remainder = full_name[len(library_full_name) :].lstrip("\\")
196        segments = [seg.rsplit(".", 1)[0] for seg in remainder.split("\\") if seg]
197        return "\\".join(self.gui_name(seg) for seg in segments)

200    # diagnostics
201    # ------------------------------------------------------------------ #
202    def unmapped_system_folders(self, within: str = "Templates") -> list[str]:
203        """`loc_name`s of system folders (below `within`) that have no translation.
204
205        A non-empty result means the localisation table and this PowerFactory
206        version have drifted - navigation still works via literal `loc_name`s.
207
208        Args:
209            within: GUI path of the subtree to check. Defaults to `"Templates"`
210                (the branch powfacpy uses). Pass `""` for the whole library
211                (slow - the global library has thousands of folders).
212        """
213        translations = self.folder_name_translations
214        start = self.get_object(within) if within else self.root
215        unmapped: list[str] = []
216
217        def walk(folder: PFGeneral) -> None:
218            for child in folder.GetContents("*.IntFolder"):
219                loc_name = child.loc_name
220                if (
221                    child.HasAttribute("iopt_sys")
222                    and child.GetAttribute("iopt_sys") == 1
223                    and " " not in loc_name  # coded keys never contain spaces
224                    and loc_name not in translations
225                ):
226                    unmapped.append(loc_name)
227                walk(child)
228
229        walk(start)
230        return unmapped
231
232    # ------------------------------------------------------------------ #
233    # internals
234    # ------------------------------------------------------------------ #
235    def _child_by_name(self, parent: PFGeneral, name: str) -> PFGeneral | None:
236        matches = [
237            child
238            for child in parent.GetContents()
239            if self.gui_name(child) == name or child.loc_name == name
240        ]
241        if len(matches) > 1:
242            raise PFInterfaceError(
243                f"'{name}' is ambiguous under '{self.gui_path(parent)}' "
244                f"({[c.GetFullName() for c in matches]})."
245            )
246        return matches[0] if matches else None
247
248    def _get_of_class(
249        self,
250        display_path: str,
251        classes: tuple[str, ...],
252        error_if_non_existent: bool,
253    ) -> PFGeneral | None:
254        obj = self.get_object(display_path, error_if_non_existent)
255        if obj is None:
256            return None
257        if obj.GetClassName() not in classes:
258            raise PFInterfaceError(
259                f"'{display_path}' is a {obj.GetClassName()}, expected one of {classes}."
260            )
261        return obj
