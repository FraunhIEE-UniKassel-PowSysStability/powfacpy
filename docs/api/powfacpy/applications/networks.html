 28    def get_vacant_cubicle_of_terminal(
 29        self,
 30        terminal: ElmTerm | str,
 31        new_cubicle_name: str | None = None,
 32    ) -> StaCubic:
 33        """Return a free cubicle (`StaCubic`) of a terminal, creating one if needed.
 34
 35        The first cubicle whose `obj_id` is ``None`` (nothing connected) is
 36        returned. If every cubicle is occupied a new one is created; its name is
 37        `new_cubicle_name`, or ``Cub_<n>`` with the lowest free ``n``.
 38
 39        Args:
 40            terminal: the `ElmTerm` (or its path) to search.
 41            new_cubicle_name: name to set on the found or created cubicle
 42                (default: keep the existing name / auto-name a new one).
 43
 44        Returns:
 45            StaCubic: the vacant cubicle.
 46        """
 47        terminal = self.act_prj._handle_single_pf_object_or_path_input(terminal)
 48        cubicles = self.get_cubicles_of_terminal(terminal)
 49        for cubicle in cubicles:
 50            if cubicle.obj_id == None:
 51                if new_cubicle_name:
 52                    cubicle.loc_name = new_cubicle_name
 53                return cubicle
 54        if not new_cubicle_name:
 55            cubicle_already_exists = True
 56            cub_num = len(cubicles) + 1
 57            while cubicle_already_exists is not None:
 58                new_cubicle_name = "Cub_" + str(cub_num)
 59                cubicle_already_exists = self.act_prj.get_unique_obj(
 60                    new_cubicle_name,
 61                    parent_folder=terminal,
 62                    error_if_non_existent=False,
 63                )
 64                cub_num += 1
 65        return self.act_prj.create_in_folder(
 66            new_cubicle_name + ".StaCubic", terminal, overwrite=False
 67        )

129    def copy_grid(
130        self,
131        grid_or_path: ElmNet | str,
132        target_folder: PFGeneral | str,
133        new_name: str,
134        parent_folder: PFGeneral | str | None = None,
135        error_if_non_existent: bool = True,
136        overwrite: bool = True,
137        use_existing: bool = False,
138        copy_diagram: bool = True,
139        deactivate: bool = False,
140    ) -> ElmNet:
141        """Copy a grid (`ElmNet`), and by default its single-line diagram too.
142
143        PowerFactory does not do this cleanly on its own:
144
145        - ``AddCopy`` silently does nothing when the source grid is **active**.
146          If the first attempt returns nothing, the active study case is briefly
147          deactivated for a retry and reactivated afterwards (deactivating the
148          *case*, not the grid - toggling an individual grid corrupts the RMS
149          calculation state for the rest of the session).
150        - the element ``bus1`` / ``bus2`` *views* only exist while a grid is
151          active (the connections themselves are stored on the cubicles). The
152          copy is therefore always **activated once and flushed** so PowerFactory builds and commits its topology; pass ``deactivate=True`` to switch it back off afterwards.
153        - the diagram (`IntGrfnet`) is not copied with the grid, and its
154          graphical objects (`IntGrf`) keep pointing at the *original*
155          elements; this method copies the diagram and re-points them at the
156          copied elements.
157
158        Graphical objects that cannot be re-pointed - decorations without a
159        data object, and elements that live in an active variation rather than
160        in the grid itself - are left untouched and counted in a warning.
161
162        Args:
163            grid_or_path: the `ElmNet` to copy (or its path).
164            target_folder: folder for the copied grid.
165            new_name: `loc_name` of the copy (also used for the copied diagram).
166            parent_folder: folder a string `grid_or_path` is resolved against.
167            error_if_non_existent: passed to the object lookup.
168            overwrite: overwrite an existing grid / diagram of the same name.
169            use_existing: return an existing copy instead of making a new one.
170            copy_diagram: also copy and re-link the single-line diagram.
171            deactivate: deactivate the copy again after the mandatory
172                activate-and-flush (default: leave it active).
173
174        Returns:
175            ElmNet: the copied grid.
176        """
177        grid_to_be_copied = self.act_prj._handle_single_pf_object_or_path_input(
178            grid_or_path
179        )
180        target_folder = self.act_prj._handle_single_pf_object_or_path_input(
181            target_folder
182        )
183        if use_existing:
184            existing = self.act_prj.get_unique_obj(
185                f"{new_name}.ElmNet",
186                parent_folder=target_folder,
187                error_if_non_existent=False,
188            )
189            if existing is not None:
190                return existing
191        new_grid = self._copy_grid_object(
192            grid_to_be_copied,
193            target_folder,
194            new_name,
195            parent_folder=parent_folder,
196            error_if_non_existent=error_if_non_existent,
197            overwrite=overwrite,
198            use_existing=use_existing,
199        )
200        source_diagram = grid_to_be_copied.pDiagram
201        if copy_diagram and source_diagram is not None:
202            # `new_grid` is always freshly copied at this point (a reused grid
203            # returned early above), so it needs a fresh diagram: `use_existing`
204            # here would re-link it to a stale, possibly empty `IntGrfnet` left
205            # in the diagrams folder by an earlier run.
206            self._copy_and_relink_diagram(
207                grid_to_be_copied,
208                new_grid,
209                source_diagram,
210                new_name,
211                overwrite=True,
212                use_existing=False,
213            )
214        # always activate once + flush so PowerFactory builds the copy's topology
215        # and commits it; the element ``bus1`` / ``bus2`` views only exist while
216        # a grid is active (the connections themselves are stored on the cubicles)
217        new_grid.Activate()
218        self.act_prj.app.WriteChangesToDb()
219        if deactivate:
220            new_grid.Deactivate()
221        return new_grid

223    def _copy_grid_object(
224        self,
225        grid: ElmNet,
226        target_folder: PFGeneral,
227        new_name: str,
228        *,
229        parent_folder: PFGeneral | str | None,
230        error_if_non_existent: bool,
231        overwrite: bool,
232        use_existing: bool,
233    ) -> ElmNet:
234        """Copy just the `ElmNet` object (no diagram).
235
236        `AddCopy` returns nothing when the source grid is active; if that
237        happens the active study case is deactivated for one retry and
238        reactivated afterwards (never the grid itself - see `copy_grid`).
239
240        Raises:
241            PFInterfaceError: if the copy fails even with the study case off, or
242                there is no active study case to deactivate for the retry.
243        """
244
245        def _copy() -> ElmNet | None:
246            return self.act_prj.copy_single_obj(
247                grid,
248                target_folder,
249                overwrite=overwrite,
250                use_existing=use_existing,
251                new_name=new_name,
252                parent_folder=parent_folder,
253                error_if_non_existent=error_if_non_existent,
254            )
255
256        new_grid = _copy()
257        if new_grid is not None:
258            return new_grid
259        # AddCopy no-ops on an active grid -> retry with the study case off
260        active_case = self.act_prj.app.GetActiveStudyCase()
261        if active_case is None:
262            raise PFInterfaceError(
263                f"copying grid '{grid.loc_name}' failed and there is no active "
264                "study case to deactivate for a retry."
265            )
266        active_case.Deactivate()
267        try:
268            new_grid = _copy()
269        finally:
270            active_case.Activate()
271        if new_grid is None:
272            raise PFInterfaceError(
273                f"copying grid '{grid.loc_name}' failed even with the study "
274                "case deactivated."
275            )
276        return new_grid

278    def _copy_and_relink_diagram(
279        self,
280        source_grid: ElmNet,
281        new_grid: ElmNet,
282        source_diagram: PFGeneral,
283        new_name: str,
284        *,
285        overwrite: bool,
286        use_existing: bool,
287    ) -> PFGeneral:
288        """Copy the single-line diagram (`IntGrfnet`) of a grid and re-point its
289        graphical objects (`IntGrf`) from the source elements to the copied ones.
290
291        Graphical objects that cannot be re-pointed (decorations without a data
292        object, elements added by an active variation) are left untouched and
293        reported in a `RuntimeWarning`. The new diagram is cross-linked with
294        `new_grid` (`pDataFolder` / `pDiagram`).
295
296        Returns:
297            PFGeneral: the copied `IntGrfnet`.
298        """
299        new_diagram = self.act_prj.copy_single_obj(
300            source_diagram,
301            source_diagram.GetParent(),
302            new_name=new_name,
303            overwrite=overwrite,
304            use_existing=use_existing,
305        )
306        unresolved = 0
307        for graphical_object in self.act_prj.get_obj(
308            "*.IntGrf",
309            parent_folder=new_diagram,
310            include_subfolders=True,
311            error_if_non_existent=False,
312        ):
313            element = graphical_object.pDataObj
314            inside = element is not None and element.GetFullName().startswith(
315                source_grid.GetFullName() + "\\"
316            )
317            if not inside:
318                unresolved += element is not None
319                continue
320            path_in_grid = self._path_inside_grid(source_grid, element)
321            copied_element = self.act_prj.get_unique_obj(
322                path_in_grid, parent_folder=new_grid, error_if_non_existent=False
323            )
324            if copied_element is not None:
325                graphical_object.pDataObj = copied_element
326            else:
327                unresolved += 1
328        if unresolved:
329            warn(
330                f"copy_grid: {unresolved} graphical object(s) in the copied "
331                f"diagram of '{new_name}' could not be re-linked (decorations, "
332                "or elements added by an active variation) and still point at "
333                f"'{source_grid.loc_name}'.",
334                RuntimeWarning,
335            )
336        new_diagram.pDataFolder = new_grid
337        new_grid.pDiagram = new_diagram
338        return new_diagram

360    def get_graphical_objects(
361        self,
362        elements: Iterable[PFGeneral],
363        *,
364        diagram: PFGeneral | str | None = None,
365    ) -> dict[PFGeneral, list[PFGeneral]]:
366        """Map each network element to the graphical objects (`IntGrf`) that draw it.
367
368        An element can be drawn in several diagrams and, rarely, more than once in one diagram, so the value is a list (usually of length one).
369
370        Args:
371            elements: the network elements (`ElmTerm`, `ElmLne`, ...) to look up.
372            diagram: which diagram to search. ``None`` (default) searches the
373                ``pDiagram`` of each element's parent grid (`get_parent_grid`); an
374                `IntGrfnet` or its path restricts the search to that one diagram.
375
376        Returns:
377            ``{element: [IntGrf, ...]}`` - every element in ``elements`` is a key;
378            the list is empty when the element is not drawn in the searched
379            diagram(s).
380        """
381        elements = list(elements)
382        index: dict[PFGeneral, list[PFGeneral]] = {}
383        for diagram_object in self._resolve_diagrams(diagram, elements):
384            for graphical_object in diagram_object.GetContents("*.IntGrf", 1):
385                data_object = graphical_object.pDataObj
386                if data_object is not None:
387                    index.setdefault(data_object, []).append(graphical_object)
388        return {element: index.get(element, []) for element in elements}

393    def get_positions(
394        self,
395        elements: Iterable[PFGeneral],
396        *,
397        diagram: PFGeneral | str | None = None,
398        aggregate: str = "first",
399    ) -> dict[PFGeneral, tuple[float, float]]:
400        """Canvas centre ``(x, y)`` of each element's graphical object.
401
402        ``x`` grows to the right; ``y`` follows the diagram's own convention (in
403        PowerFactory single-line diagrams larger ``y`` is higher on the page).
404        Built on `get_graphical_objects`; intended for node-like elements
405        (`ElmTerm`) - a line's graphic is a polyline and may not have a centre.
406
407        Args:
408            elements: network elements to locate.
409            diagram: see `get_graphical_objects`.
410            aggregate: what to return when an element has several graphical
411                objects - ``"first"`` or ``"mean"`` (centroid).
412
413        Returns:
414            ``{element: (x, y)}`` - elements with no usable graphical object are
415            omitted and reported in a warning.
416        """
417        if aggregate not in ("first", "mean"):
418            raise PFInterfaceError("aggregate must be 'first' or 'mean'")
419        graphical_objects = self.get_graphical_objects(elements, diagram=diagram)
420        positions: dict[PFGeneral, tuple[float, float]] = {}
421        without_position: list[PFGeneral] = []
422        for element, objects in graphical_objects.items():
423            centres = [
424                self._graphical_object_centre(graphical_object)
425                for graphical_object in objects
426            ]
427            centres = [centre for centre in centres if centre is not None]
428            if not centres:
429                without_position.append(element)
430            elif aggregate == "first" or len(centres) == 1:
431                positions[element] = centres[0]
432            else:
433                positions[element] = (
434                    sum(x for x, _ in centres) / len(centres),
435                    sum(y for _, y in centres) / len(centres),
436                )
437        if without_position:
438            names = [element.loc_name for element in without_position[:5]]
439            warn(
440                f"get_positions: no graphical object for "
441                f"{len(without_position)} element(s): {names}"
442                + (" ..." if len(without_position) > 5 else ""),
443                RuntimeWarning,
444            )
445        return positions

477    def rename_branches_after_terminals(
478        self,
479        branches: Iterable[PFGeneral],
480        *,
481        prefix: str = "L",
482        separator: str = "-",
483        short_name: Callable[[str], str] | None = None,
484        order_key: Callable[[str], object] | None = None,
485    ) -> dict[PFGeneral, str]:
486        """Rename branch elements after the two terminals they connect.
487
488        ``<prefix><sep><a><sep><b>``, where ``a`` and ``b`` are the names of the
489        terminals on the branch's two ends, passed through ``short_name`` and
490        ordered by ``order_key``. Parallel branches that would collide get a
491        ``<sep><n>`` suffix.
492
493        Args:
494            branches: branch elements - anything with two connected bus attributes
495                among ``bus1`` / ``bus2`` / ``bushv`` / ``busmv`` / ``buslv``
496                (`ElmLne`, `ElmTr2`, `ElmCoup`, `ElmZpu`, ...).
497            prefix: leading token (``"L"`` for lines, ``"T"`` for transformers).
498            separator: token separator.
499            short_name: maps a terminal ``loc_name`` to the token used in the
500                branch name (default: the terminal name unchanged).
501            order_key: sort key applied to the two tokens so the result does not
502                depend on PowerFactory's ``bus1`` / ``bus2`` order (default: keep
503                the branch's own order; pass ``str`` for alphabetical).
504
505        Returns:
506            ``{branch: new loc_name}``.
507        """
508        short_name = short_name or (lambda name: name)
509        used: dict[str, int] = {}
510        renamed: dict[PFGeneral, str] = {}
511        for branch in branches:
512            tokens = [
513                short_name(terminal.loc_name)
514                for terminal in self._branch_terminals(branch)
515            ]
516            if order_key is not None:
517                tokens = sorted(tokens, key=order_key)
518            name = separator.join([prefix, *tokens])
519            occurrence = used.get(name, 0)
520            used[name] = occurrence + 1
521            if occurrence:
522                name = f"{name}{separator}{occurrence}"
523            branch.loc_name = name
524            renamed[branch] = name
525        return renamed

527    def rename_children(
528        self,
529        container: PFGeneral | str,
530        *,
531        prefix: str = "",
532        suffix: str = "",
533        include_subfolders: bool = False,
534    ) -> dict[PFGeneral, str]:
535        """Add a `prefix` and/or `suffix` to the `loc_name` of a container's children.
536
537        Handy when several copies of the same grid are active in one study case:

547            prefix: prepended to every child `loc_name`.

557        )
558        children = (
559            container.GetContents("*", 1)
560            if include_subfolders
561            else container.GetContents()
562        )
563        renamed: dict[PFGeneral, str] = {}
564        for child in children:

566            if new_name is not None:

571    def rename_drawn_elements(
572        self,
573        grid_or_diagram: PFGeneral | str,
574        *,
575        prefix: str = "",
576        suffix: str = "",
577    ) -> dict[PFGeneral, str]:
578        """Add a `prefix` / `suffix` to the `loc_name` of every element drawn in a diagram.
579
580        Iterates the graphical objects (`IntGrf`) of the single-line diagram and
581        renames each one's data object (`pDataObj`) - i.e. only the terminals,
582        branches, machines, loads, ... that actually appear on the drawing.
583        Cubicles, folders, composite-model internals and decorations without a
584        data object are left alone. Idempotent.
585
586        Args:
587            grid_or_diagram: an `ElmNet` (its `pDiagram` is used) or an
588                `IntGrfnet` directly (or a path to either).
589            prefix: prepended to every drawn element's `loc_name`.
590            suffix: appended to every drawn element's `loc_name`.
591
592        Returns:
593            ``{element: new loc_name}`` for the elements that were renamed.
594        """
595        obj = self.act_prj._handle_single_pf_object_or_path_input(
596            grid_or_diagram
597        )
598        diagram = obj.pDiagram if obj.GetClassName() == "ElmNet" else obj
599        if diagram is None:
600            raise PFInterfaceError(
601                f"'{obj.loc_name}' has no single-line diagram (pDiagram)."
602            )
603        renamed: dict[PFGeneral, str] = {}
604        for graphical_object in diagram.GetContents("*.IntGrf", 1):
605            element = graphical_object.pDataObj
606            if element is None or element in renamed:
607                continue
608            new_name = _affix(element.loc_name, prefix, suffix)
609            if new_name is not None:
610                element.loc_name = new_name
611                renamed[element] = new_name

628        self, branch: PFGeneral | str, terminal: ElmTerm | str
629    ) -> StaCubic:

631

633        `bus2` / `bushv` / `busmv` / `buslv` equal to ``None``) - e.g. a breaker

635        vacant cubicle of `terminal` is used (a new one is created if none is
636        free).
637

639            branch: the branch element (or its path) to connect.
640            terminal: the `ElmTerm` (or its path) to connect it to.
641
642        Returns:
643            StaCubic: the cubicle of `terminal` the branch was wired into.
644        """
645        branch = self.act_prj._handle_single_pf_object_or_path_input(branch)
646        terminal = self.act_prj._handle_single_pf_object_or_path_input(terminal)
647        open_attributes = [
648            attribute
649            for attribute in _BRANCH_BUS_ATTRIBUTES
650            if branch.HasAttribute(attribute)
651            and branch.GetAttribute(attribute) is None
652        ]
653        if len(open_attributes) != 1:
654            raise PFInterfaceError(
655                f"'{branch.loc_name}' ({branch.GetClassName()}) has "
656                f"{len(open_attributes)} disconnected cubicle(s) - `connect` "
657                "needs exactly one; define an unambiguous connection element."
658            )
659        cubicle = self.get_vacant_cubicle_of_terminal(terminal)
660        branch.SetAttribute(open_attributes[0], cubicle)
661        return cubicle

 30        self,

 32        new_cubicle_name: str | None = None,

 35

130    def copy_grid(
131        self,
132        grid_or_path: ElmNet | str,
133        target_folder: PFGeneral | str,
134        new_name: str,
135        parent_folder: PFGeneral | str | None = None,
136        error_if_non_existent: bool = True,
137        overwrite: bool = True,
138        use_existing: bool = False,
139        copy_diagram: bool = True,
140        deactivate: bool = False,
141    ) -> ElmNet:
142        """Copy a grid (`ElmNet`), and by default its single-line diagram too.
143
144        PowerFactory does not do this cleanly on its own:
145
146        - ``AddCopy`` silently does nothing when the source grid is **active**.
147          If the first attempt returns nothing, the active study case is briefly
148          deactivated for a retry and reactivated afterwards (deactivating the
149          *case*, not the grid - toggling an individual grid corrupts the RMS
150          calculation state for the rest of the session).
151        - the element ``bus1`` / ``bus2`` *views* only exist while a grid is
152          active (the connections themselves are stored on the cubicles). The
153          copy is therefore always **activated once and flushed** so PowerFactory builds and commits its topology; pass ``deactivate=True`` to switch it back off afterwards.
154        - the diagram (`IntGrfnet`) is not copied with the grid, and its
155          graphical objects (`IntGrf`) keep pointing at the *original*
156          elements; this method copies the diagram and re-points them at the
157          copied elements.
158
159        Graphical objects that cannot be re-pointed - decorations without a
160        data object, and elements that live in an active variation rather than
161        in the grid itself - are left untouched and counted in a warning.
162
163        Args:
164            grid_or_path: the `ElmNet` to copy (or its path).
165            target_folder: folder for the copied grid.
166            new_name: `loc_name` of the copy (also used for the copied diagram).
167            parent_folder: folder a string `grid_or_path` is resolved against.
168            error_if_non_existent: passed to the object lookup.
169            overwrite: overwrite an existing grid / diagram of the same name.
170            use_existing: return an existing copy instead of making a new one.
171            copy_diagram: also copy and re-link the single-line diagram.
172            deactivate: deactivate the copy again after the mandatory
173                activate-and-flush (default: leave it active).
174
175        Returns:
176            ElmNet: the copied grid.
177        """
178        grid_to_be_copied = self.act_prj._handle_single_pf_object_or_path_input(
179            grid_or_path
180        )
181        target_folder = self.act_prj._handle_single_pf_object_or_path_input(
182            target_folder
183        )
184        if use_existing:
185            existing = self.act_prj.get_unique_obj(
186                f"{new_name}.ElmNet",
187                parent_folder=target_folder,
188                error_if_non_existent=False,
189            )
190            if existing is not None:
191                return existing
192        new_grid = self._copy_grid_object(
193            grid_to_be_copied,
194            target_folder,
195            new_name,
196            parent_folder=parent_folder,
197            error_if_non_existent=error_if_non_existent,
198            overwrite=overwrite,
199            use_existing=use_existing,
200        )
201        source_diagram = grid_to_be_copied.pDiagram
202        if copy_diagram and source_diagram is not None:
203            # `new_grid` is always freshly copied at this point (a reused grid
204            # returned early above), so it needs a fresh diagram: `use_existing`
205            # here would re-link it to a stale, possibly empty `IntGrfnet` left
206            # in the diagrams folder by an earlier run.
207            self._copy_and_relink_diagram(
208                grid_to_be_copied,
209                new_grid,
210                source_diagram,
211                new_name,
212                overwrite=True,
213                use_existing=False,
214            )
215        # always activate once + flush so PowerFactory builds the copy's topology
216        # and commits it; the element ``bus1`` / ``bus2`` views only exist while
217        # a grid is active (the connections themselves are stored on the cubicles)
218        new_grid.Activate()
219        self.act_prj.app.WriteChangesToDb()
220        if deactivate:
221            new_grid.Deactivate()
222        return new_grid

224    def _copy_grid_object(
225        self,
226        grid: ElmNet,
227        target_folder: PFGeneral,
228        new_name: str,
229        *,
230        parent_folder: PFGeneral | str | None,
231        error_if_non_existent: bool,
232        overwrite: bool,
233        use_existing: bool,
234    ) -> ElmNet:
235        """Copy just the `ElmNet` object (no diagram).
236
237        `AddCopy` returns nothing when the source grid is active; if that
238        happens the active study case is deactivated for one retry and
239        reactivated afterwards (never the grid itself - see `copy_grid`).
240
241        Raises:
242            PFInterfaceError: if the copy fails even with the study case off, or
243                there is no active study case to deactivate for the retry.
244        """
245
246        def _copy() -> ElmNet | None:
247            return self.act_prj.copy_single_obj(
248                grid,
249                target_folder,
250                overwrite=overwrite,
251                use_existing=use_existing,
252                new_name=new_name,
253                parent_folder=parent_folder,
254                error_if_non_existent=error_if_non_existent,
255            )
256
257        new_grid = _copy()
258        if new_grid is not None:
259            return new_grid
260        # AddCopy no-ops on an active grid -> retry with the study case off
261        active_case = self.act_prj.app.GetActiveStudyCase()
262        if active_case is None:
263            raise PFInterfaceError(
264                f"copying grid '{grid.loc_name}' failed and there is no active "
265                "study case to deactivate for a retry."
266            )
267        active_case.Deactivate()
268        try:
269            new_grid = _copy()
270        finally:
271            active_case.Activate()
272        if new_grid is None:
273            raise PFInterfaceError(
274                f"copying grid '{grid.loc_name}' failed even with the study "
275                "case deactivated."
276            )
277        return new_grid

279    def _copy_and_relink_diagram(
280        self,
281        source_grid: ElmNet,
282        new_grid: ElmNet,
283        source_diagram: PFGeneral,
284        new_name: str,
285        *,
286        overwrite: bool,
287        use_existing: bool,
288    ) -> PFGeneral:
289        """Copy the single-line diagram (`IntGrfnet`) of a grid and re-point its
290        graphical objects (`IntGrf`) from the source elements to the copied ones.
291
292        Graphical objects that cannot be re-pointed (decorations without a data
293        object, elements added by an active variation) are left untouched and
294        reported in a `RuntimeWarning`. The new diagram is cross-linked with
295        `new_grid` (`pDataFolder` / `pDiagram`).
296
297        Returns:
298            PFGeneral: the copied `IntGrfnet`.
299        """
300        new_diagram = self.act_prj.copy_single_obj(
301            source_diagram,
302            source_diagram.GetParent(),
303            new_name=new_name,
304            overwrite=overwrite,
305            use_existing=use_existing,
306        )
307        unresolved = 0
308        for graphical_object in self.act_prj.get_obj(
309            "*.IntGrf",
310            parent_folder=new_diagram,
311            include_subfolders=True,
312            error_if_non_existent=False,
313        ):
314            element = graphical_object.pDataObj
315            inside = element is not None and element.GetFullName().startswith(
316                source_grid.GetFullName() + "\\"
317            )
318            if not inside:
319                unresolved += element is not None
320                continue
321            path_in_grid = self._path_inside_grid(source_grid, element)
322            copied_element = self.act_prj.get_unique_obj(
323                path_in_grid, parent_folder=new_grid, error_if_non_existent=False
324            )
325            if copied_element is not None:
326                graphical_object.pDataObj = copied_element
327            else:
328                unresolved += 1
329        if unresolved:
330            warn(
331                f"copy_grid: {unresolved} graphical object(s) in the copied "
332                f"diagram of '{new_name}' could not be re-linked (decorations, "
333                "or elements added by an active variation) and still point at "
334                f"'{source_grid.loc_name}'.",
335                RuntimeWarning,
336            )
337        new_diagram.pDataFolder = new_grid
338        new_grid.pDiagram = new_diagram
339        return new_diagram

394    def get_positions(
395        self,
396        elements: Iterable[PFGeneral],
397        *,
398        diagram: PFGeneral | str | None = None,
399        aggregate: str = "first",
400    ) -> dict[PFGeneral, tuple[float, float]]:
401        """Canvas centre ``(x, y)`` of each element's graphical object.
402
403        ``x`` grows to the right; ``y`` follows the diagram's own convention (in
404        PowerFactory single-line diagrams larger ``y`` is higher on the page).
405        Built on `get_graphical_objects`; intended for node-like elements
406        (`ElmTerm`) - a line's graphic is a polyline and may not have a centre.
407
408        Args:
409            elements: network elements to locate.
410            diagram: see `get_graphical_objects`.
411            aggregate: what to return when an element has several graphical
412                objects - ``"first"`` or ``"mean"`` (centroid).
413
414        Returns:
415            ``{element: (x, y)}`` - elements with no usable graphical object are
416            omitted and reported in a warning.
417        """
418        if aggregate not in ("first", "mean"):
419            raise PFInterfaceError("aggregate must be 'first' or 'mean'")
420        graphical_objects = self.get_graphical_objects(elements, diagram=diagram)
421        positions: dict[PFGeneral, tuple[float, float]] = {}
422        without_position: list[PFGeneral] = []
423        for element, objects in graphical_objects.items():
424            centres = [
425                self._graphical_object_centre(graphical_object)
426                for graphical_object in objects
427            ]
428            centres = [centre for centre in centres if centre is not None]
429            if not centres:
430                without_position.append(element)
431            elif aggregate == "first" or len(centres) == 1:
432                positions[element] = centres[0]
433            else:
434                positions[element] = (
435                    sum(x for x, _ in centres) / len(centres),
436                    sum(y for _, y in centres) / len(centres),
437                )
438        if without_position:
439            names = [element.loc_name for element in without_position[:5]]
440            warn(
441                f"get_positions: no graphical object for "
442                f"{len(without_position)} element(s): {names}"
443                + (" ..." if len(without_position) > 5 else ""),
444                RuntimeWarning,
445            )
446        return positions

478    def rename_branches_after_terminals(
479        self,
480        branches: Iterable[PFGeneral],
481        *,
482        prefix: str = "L",
483        separator: str = "-",
484        short_name: Callable[[str], str] | None = None,
485        order_key: Callable[[str], object] | None = None,
486    ) -> dict[PFGeneral, str]:
487        """Rename branch elements after the two terminals they connect.
488
489        ``<prefix><sep><a><sep><b>``, where ``a`` and ``b`` are the names of the
490        terminals on the branch's two ends, passed through ``short_name`` and
491        ordered by ``order_key``. Parallel branches that would collide get a
492        ``<sep><n>`` suffix.
493
494        Args:
495            branches: branch elements - anything with two connected bus attributes
496                among ``bus1`` / ``bus2`` / ``bushv`` / ``busmv`` / ``buslv``
497                (`ElmLne`, `ElmTr2`, `ElmCoup`, `ElmZpu`, ...).
498            prefix: leading token (``"L"`` for lines, ``"T"`` for transformers).
499            separator: token separator.
500            short_name: maps a terminal ``loc_name`` to the token used in the
501                branch name (default: the terminal name unchanged).
502            order_key: sort key applied to the two tokens so the result does not
503                depend on PowerFactory's ``bus1`` / ``bus2`` order (default: keep
504                the branch's own order; pass ``str`` for alphabetical).
505
506        Returns:
507            ``{branch: new loc_name}``.
508        """
509        short_name = short_name or (lambda name: name)
510        used: dict[str, int] = {}
511        renamed: dict[PFGeneral, str] = {}
512        for branch in branches:
513            tokens = [
514                short_name(terminal.loc_name)
515                for terminal in self._branch_terminals(branch)
516            ]
517            if order_key is not None:
518                tokens = sorted(tokens, key=order_key)
519            name = separator.join([prefix, *tokens])
520            occurrence = used.get(name, 0)
521            used[name] = occurrence + 1
522            if occurrence:
523                name = f"{name}{separator}{occurrence}"
524            branch.loc_name = name
525            renamed[branch] = name
526        return renamed

529        self,
530        container: PFGeneral | str,
531        *,
532        prefix: str = "",
533        suffix: str = "",
534        include_subfolders: bool = False,
535    ) -> dict[PFGeneral, str]:
536        """Add a `prefix` and/or `suffix` to the `loc_name` of a container's children.
537
538        Handy when several copies of the same grid are active in one study case:
539        without a distinguishing affix their elements all share names like
540        ``Terminal`` / ``General Load`` and cannot be told apart by

551                composite models (default: only the direct children).

504                the branch's own order; pass ``str`` for alphabetical).
505
506        Returns:
507            ``{branch: new loc_name}``.
508        """
509        short_name = short_name or (lambda name: name)
510        used: dict[str, int] = {}
511        renamed: dict[PFGeneral, str] = {}
512        for branch in branches:
513            tokens = [
514                short_name(terminal.loc_name)
515                for terminal in self._branch_terminals(branch)
516            ]
517            if order_key is not None:
518                tokens = sorted(tokens, key=order_key)
519            name = separator.join([prefix, *tokens])
520            occurrence = used.get(name, 0)
521            used[name] = occurrence + 1
522            if occurrence:
523                name = f"{name}{separator}{occurrence}"
524            branch.loc_name = name
525            renamed[branch] = name
526        return renamed
