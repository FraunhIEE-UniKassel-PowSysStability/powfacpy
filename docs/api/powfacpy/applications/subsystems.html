139    def get_power_margins(self, execute_load_flow: bool = True) -> pd.DataFrame:
140        """Upward / downward active-power headroom [MW] per internal generator.
141
142        `upward_margin = P_max - P`, `downward_margin = P - P_min`, where `P` is
143        the load-flow active power and `P_max` / `P_min` are the operational
144        limits configured on the element (`P_max` / `Pmax_uc`, `P_min` /
145        `Pmin_uc`). The limits are only as meaningful as the model - a machine
146        dispatched outside its configured limits yields a negative margin.
147
148        Args:
149            execute_load_flow: run a load flow first so `P` is up to date.
150
151        Returns:
152            pd.DataFrame: one row per generator with columns ``name``, ``class``,
153            ``P_MW``, ``P_max_MW``, ``P_min_MW``, ``upward_margin_MW``,
154            ``downward_margin_MW``.
155        """
156        if execute_load_flow:
157            self.parent.act_prj.execute_load_flow()
158        rows = []
159        for gen in self._dispatchable_generators():
160            p = _first_attr(gen, "m:P:bus1", "pgini", default=np.nan)
161            p_max = _first_attr(gen, "P_max", "Pmax_uc", "Pmax", default=np.nan)
162            p_min = _first_attr(gen, "P_min", "Pmin_uc", "Pmin", default=0.0)
163            rows.append(
164                {
165                    "name": gen.loc_name,
166                    "class": gen.GetClassName(),
167                    "P_MW": p,
168                    "P_max_MW": p_max,
169                    "P_min_MW": p_min,
170                    "upward_margin_MW": p_max - p,
171                    "downward_margin_MW": p - p_min,
172                }
173            )
174        return pd.DataFrame(
175            rows,
176            columns=[
177                "name",
178                "class",
179                "P_MW",
180                "P_max_MW",
181                "P_min_MW",
182                "upward_margin_MW",
183                "downward_margin_MW",
184            ],
185        )

193    def create_secondary_controller(
194        self,
195        name: str | None = None,
196        parent_folder: PFGeneral | None = None,
197        *,
198        synchronous_machines: bool = False,
199        batteries: bool = False,
200        wind: bool = False,
201        pv: bool = False,
202        attr: dict | None = None,
203    ) -> ElmSecctrl:
204        """Create a secondary (power-frequency) controller (`ElmSecctrl`) for the subsystem.
205
206        Thin wrapper around `StaticCalc.create_secondary_controller`. Set automatically so the controller regulates this subsystem's own interchange: `iexchange` ("Exchange for", -> "Zone" or "Area") and `pPmeas` ("Boundary/Zone/Area", -> the subsystem's grouping object). Everything else (`i_net`, `psetp`, `Kpf`, ...) is left at its PowerFactory default - configure it yourself or pass it via `attr`.
207
208        The controlled machines (`psym`) are collected from the subsystem's internal units, one category per flag that is set.
209
210        Args:
211            name: `loc_name` of the controller. Defaults to ``"<subsystem> secondary controller"``.
212
213            parent_folder: folder to create it in. Defaults to the `ElmNet` of the subsystem's first internal terminal.
214
215            synchronous_machines: add the internal synchronous generators (`get_internal_sg`) to `psym`.
216
217            batteries: add the internal battery storage units (`get_internal_bess`).
218
219            wind: add the internal wind units (`get_internal_wind`).
220
221            pv: add the internal PV units (`get_internal_pv`).
222
223            attr: further attributes (name -> value) to set on the `ElmSecctrl`.
224
225        Returns:
226            ElmSecctrl: the created controller.
227        """
228        from powfacpy.applications.static_calc import StaticCalc
229
230        subsystem = self.parent
231        controlled: list[PFGeneral] = []
232        if synchronous_machines:
233            controlled += subsystem.get_internal_sg()
234        if batteries:
235            controlled += subsystem.get_internal_bess()
236        if wind:
237            controlled += subsystem.get_internal_wind()
238        if pv:
239            controlled += subsystem.get_internal_pv()
240
241        if name is None:
242            name = f"{subsystem.name} secondary controller"
243        if parent_folder is None:
244            parent_folder = self._grid_for_new_objects()
245
246        attributes = {
247            "iexchange": self._EXCHANGE_FOR[subsystem._obj.GetClassName()],
248            "pPmeas": subsystem._obj,
249        }
250        if attr:
251            attributes.update(attr)
252
253        return StaticCalc(subsystem.app).create_secondary_controller(
254            name=name,
255            parent_folder=parent_folder,
256            controlled_objs=controlled or None,
257            attr=attributes,
258        )
259
260    def _grid_for_new_objects(self) -> PFGeneral:
261        """The `ElmNet` to create subsystem-level objects in (grid of the first internal terminal)."""
262        act_prj = self.parent.act_prj
263        for terminal in self.parent.get_internal_elms_of_class("ElmTerm"):
264            grid = act_prj.get_upstream_obj(
265                terminal,
266                lambda x: x.GetClassName() == "ElmNet",
267                error_if_non_existent=False,
268            )
269            if grid is not None:
270                return grid
271        raise PFInterfaceError(
272            f"Could not determine a grid for '{self.parent.name}'; pass "
273            "'parent_folder' explicitly."
274        )

421class SubSystemDynamics:
422    """Dynamics-related aggregate quantities of a subsystem: inertia, power
423    margins, load / generation - relevant e.g. for frequency stability and
424    intentional islanding.
425
426    Inertia is collected per dynamic unit: directly from the type for
427    synchronous machines (`ElmSym`), and - for grid-forming converters
428    (`ElmGenstat` / `ElmVsc` / `ElmPvsys` with a grid-forming control) - via
429    `powfacpy.template_models` (the controller's template is identified and its
430    `get_kinetic_energy_MWs` used). Converters whose template is not recognised
431    are reported with `inertia_MWs = NaN`.
432    """
433
434    NOMINAL_FREQUENCY_HZ = 50.0
435
436    def __init__(self, parent: SubSystem) -> None:
437        self.parent = parent
438
439    # ------------------------------------------------------------------ #
440    # inertia
441    # ------------------------------------------------------------------ #
442    @cached_property
443    def _synchronous_machine_rows(self) -> list[dict]:
444        rows = []
445        for sm in self.parent.dynamic_models.synchronous_machines.synchronous_machines_powfacpy:
446            s = sm.ratedS
447            h = sm.get_H_in_seconds()
448            rows.append(
449                {
450                    "name": sm._obj.loc_name,
451                    "class": "ElmSym",
452                    "S_MVA": s,
453                    "H_s": h,
454                    "inertia_MWs": h * s,
455                    "grid_forming": True,
456                    "template": "ElmSym/TypSym",
457                }
458            )
459        return rows
460
461    @cached_property
462    def _converter_rows(self) -> list[dict]:
463        from powfacpy.template_models import TemplateMatcher
464
465        matcher = TemplateMatcher(app=self.parent.app)
466        rows = []
467        for conv in self.parent.get_grid_forming_converters():
468            composite_model = (
469                conv.c_pmod if conv.HasAttribute("c_pmod") else None
470            )
471            row = {
472                "name": conv.loc_name,
473                "class": conv.GetClassName(),
474                "S_MVA": conv.GetAttribute("sgn")
475                if conv.HasAttribute("sgn")
476                else np.nan,
477                "H_s": np.nan,
478                "inertia_MWs": np.nan,
479                "grid_forming": True,
480                "template": None,
481            }
482            if composite_model is not None:
483                match = matcher.identify(composite_model)
484                model = match.build(network_element=conv)
485                if model is not None:
486                    row["template"] = type(model).__name__
487                    row["H_s"] = model.get_equivalent_inertia_constant()
488                    row["inertia_MWs"] = model.get_kinetic_energy_MWs()
489                elif match.library_template_paths:
490                    # template recognised but no powfacpy class -> no inertia
491                    row["template"] = match.library_template_paths[0]
492            rows.append(row)
493        return rows
494
495    def get_inertia_details(self) -> pd.DataFrame:
496        """Per-unit inertia table (synchronous machines and grid-forming converters)."""
497        rows = self._synchronous_machine_rows + self._converter_rows
498        return pd.DataFrame(
499            rows,
500            columns=[
501                "name",
502                "class",
503                "S_MVA",
504                "H_s",
505                "inertia_MWs",
506                "grid_forming",
507                "template",
508            ],
509        )
510
511    def synchronous_machine_inertia_MWs(self) -> float:
512        return float(np.nansum([r["inertia_MWs"] for r in self._synchronous_machine_rows]))
513
514    def converter_inertia_MWs(self) -> float:
515        return float(np.nansum([r["inertia_MWs"] for r in self._converter_rows]))
516
517    def total_inertia_MWs(self) -> float:
518        """Total stored kinetic energy [MW s] (synchronous + grid-forming converter)."""
519        return self.synchronous_machine_inertia_MWs() + self.converter_inertia_MWs()
520
521    def inertia_constant_on_base(self, base_MVA: float) -> float:
522        """System inertia constant `H = sum(H_i * S_i) / base_MVA` [s]."""
523        return self.total_inertia_MWs() / base_MVA

525    # ------------------------------------------------------------------ #
526    # summary
527    # ------------------------------------------------------------------ #
528    def get_state(
529        self, execute_load_flow: bool = True, format: str | None = "pandas"
530    ) -> pd.DataFrame | dict | None:
531        """Single-row summary of the subsystem's dynamics state (inertia,
532        margins, load / generation)."""
533        load_flow = self.parent.get_load_flow_state(
534            execute_load_flow=execute_load_flow, format="dict"
535        )
536        total_generation = load_flow["total_power_generation"].real
537        state = {
538            "total_load_MW": load_flow["total_power_loads"].real,
539            "total_generation_MW": total_generation,
540            "power_exchange_MW": load_flow["total_power_exchange"].real,
541            "synchronous_inertia_MWs": self.synchronous_machine_inertia_MWs(),
542            "converter_inertia_MWs": self.converter_inertia_MWs(),
543            "total_inertia_MWs": self.total_inertia_MWs(),
544            "inertia_constant_on_generation_s": (
545                self.total_inertia_MWs() / total_generation
546                if total_generation
547                else np.nan
548            ),
549            "upward_power_margin_MW": self.parent.load_flow.upward_power_margin_MW(
550                execute_load_flow=False
551            ),
552            "downward_power_margin_MW": self.parent.load_flow.downward_power_margin_MW(
553                execute_load_flow=False
554            ),
555        }
556        if format == "dict":
557            return state
558        if format == "pandas":
559            return pd.DataFrame(state, index=[self.parent.name])
560        return None

724    def get_tie_branches(
725        self, element_classes: list[str] | None = None
726    ) -> pd.DataFrame:
727        """Branch elements connecting two or more subsystems in this container.
728
729        An element qualifies if at least two of its cubicles (`GetCubicle`)
730        are attached to terminals (`cterm`) that belong to *different*
731        subsystems of this container. This works for any PowerFactory class
732        with multiple connection points (`ElmLne`, `ElmTr2`, `ElmTr3`,
733        `ElmCoup`, `ElmSind`, ...) without hardcoding branch class names -
734        elements with a single cubicle (loads, generators, ...) can never
735        qualify, so they are excluded automatically. Restrict the search to
736        specific classes with `element_classes`, e.g. `["ElmLne"]` for tie
737        *lines* only (see also `get_tie_lines`).
738
739        This is purely topological - based on cubicle/terminal connectivity,
740        not load-flow results - so out-of-service elements and zero-flow ties
741        are still found. This differs from `is_neighbor` /
742        `get_neighboring_subsystems_in_container`, which rely on
743        `CalculateInterchangeTo` and therefore need a valid, non-zero power
744        flow to detect a connection.
745
746        Args:
747            element_classes: restrict to these PowerFactory classes. Defaults
748                to None (any class with multiple cubicles).
749
750        Returns:
751            pd.DataFrame with one row per tie element: ``element`` (the PF
752            object), ``class``, ``name`` and ``subsystems`` (tuple of the
753            >= 2 subsystem names it connects).
754        """
755        terminal_subsystem = self._terminal_to_subsystem_index()
756        rows = []
757        seen = set()
758        for subsystem in self._subsystems:
759            for terminal in subsystem.topology.terminals:
760                for cubicle in terminal.GetConnectedCubicles():
761                    elm = cubicle.obj_id
762                    if elm is None or elm in seen:
763                        continue
764                    seen.add(elm)
765                    if element_classes and elm.GetClassName() not in element_classes:
766                        continue
767                    subsystem_indices = self._connected_subsystem_indices(
768                        elm, terminal_subsystem
769                    )
770                    if len(subsystem_indices) >= 2:
771                        rows.append(
772                            {
773                                "element": elm,
774                                "class": elm.GetClassName(),
775                                "name": elm.loc_name,
776                                "subsystems": tuple(
777                                    self.names[i] for i in sorted(subsystem_indices)
778                                ),
779                            }
780                        )
781        return pd.DataFrame(rows, columns=["element", "class", "name", "subsystems"])

729

731        are attached to terminals (`cterm`) that belong to *different*
732        subsystems of this container. This works for any PowerFactory class
733        with multiple connection points (`ElmLne`, `ElmTr2`, `ElmTr3`,
734        `ElmCoup`, `ElmSind`, ...) without hardcoding branch class names -
735        elements with a single cubicle (loads, generators, ...) can never
736        qualify, so they are excluded automatically. Restrict the search to
737        specific classes with `element_classes`, e.g. `["ElmLne"]` for tie
738        *lines* only (see also `get_tie_lines`).
739

741        not load-flow results - so out-of-service elements and zero-flow ties
742        are still found. This differs from `is_neighbor` /
743        `get_neighboring_subsystems_in_container`, which rely on
744        `CalculateInterchangeTo` and therefore need a valid, non-zero power
745        flow to detect a connection.
746
747        Args:
748            element_classes: restrict to these PowerFactory classes. Defaults
749                to None (any class with multiple cubicles).
750
751        Returns:
752            pd.DataFrame with one row per tie element: ``element`` (the PF
753            object), ``class``, ``name`` and ``subsystems`` (tuple of the
754            >= 2 subsystem names it connects).
755        """
756        terminal_subsystem = self._terminal_to_subsystem_index()
757        rows = []
758        seen = set()
759        for subsystem in self._subsystems:
760            for terminal in subsystem.topology.terminals:
761                for cubicle in terminal.GetConnectedCubicles():
762                    elm = cubicle.obj_id
763                    if elm is None or elm in seen:
764                        continue
765                    seen.add(elm)
766                    if element_classes and elm.GetClassName() not in element_classes:
767                        continue
768                    subsystem_indices = self._connected_subsystem_indices(
769                        elm, terminal_subsystem
770                    )
771                    if len(subsystem_indices) >= 2:
772                        rows.append(
773                            {
774                                "element": elm,
775                                "class": elm.GetClassName(),
776                                "name": elm.loc_name,
777                                "subsystems": tuple(
778                                    self.names[i] for i in sorted(subsystem_indices)
779                                ),
780                            }
781                        )
782        return pd.DataFrame(rows, columns=["element", "class", "name", "subsystems"])
