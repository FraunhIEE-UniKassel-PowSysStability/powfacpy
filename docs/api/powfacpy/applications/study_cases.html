 83    def create_cases(
 84        self, reactivate_initially_activated_study_case: bool = True
 85    ) -> None:
 86        """Create study cases.
 87
 88        Optionally create corresponding scenarios/variations (if add_scenario_to_each_case/add_variation_to_each_case is True).
 89
 90        Iterates through all cases and creates study cases (and folders
 91        according to 'hierarchy') using parameter-value strings for the
 92        study cases (and folder names).
 93        """
 94        if not self.parameter_values:
 95            raise AttributeError("Please set the attribute 'parameter_values'.")
 96        if reactivate_initially_activated_study_case:
 97            initially_active_case = self.act_prj.app.GetActiveStudyCase()
 98        self._study_cases = []
 99        number_of_cases = len(next(iter(self.parameter_values.values())))
100        if self.base_study_case:
101            self.base_study_case = self.act_prj._handle_single_pf_object_or_path_input(
102                self.base_study_case
103            )
104        for case_num in range(number_of_cases):
105            folder_path = self.get_folder_path(case_num)
106            if self.study_cases_names:
107                case_name = self.study_cases_names[case_num]
108            else:
109                case_name = self.get_case_params_value_string(
110                    case_num, omitted_parameters=self.hierarchy
111                )
112            if self.consecutively_number_case_names:
113                case_name = str(case_num) + " " + case_name
114
115            self._study_cases.append(self._create_study_case(case_name, folder_path))
116            self.activate_grids(case_num)
117            if self.add_scenario_to_each_case:
118                scen = self._create_scenario(case_name, folder_path)
119            if self.add_variation_to_each_case:
120                self._create_variation(case_name, folder_path)
121            self._set_parameters(case_num)
122            if self.add_scenario_to_each_case:
123                scen.Save()
124        if reactivate_initially_activated_study_case and initially_active_case:
125            initially_active_case.Activate()

127    def get_folder_path(self, case_num: int) -> str | None:
128        """Get folder path (inside parent folder) of a case.
129
130        The path corresponds to parameter-value pairs specified
131        in 'self.hierarchy'.
132
133        Args:
134          case_num (int): case number
135
136        Returns:
137          str | None: path of study case or None if there is no hierarchy
138        """
139        if self.hierarchy:
140            folder_path = ""
141            for par_name in self.hierarchy:
142                parameter_value = self.get_value_of_parameter_for_case(
143                    par_name, case_num
144                )
145                if (
146                    parameter_value is not None
147                    or not self.ignore_parameters_that_are_none_in_names
148                ):
149                    add_to_string = str(parameter_value) + "\\"
150                    if not par_name in self.anonymous_parameters:
151                        add_to_string = par_name + "_" + add_to_string
152                    folder_path += add_to_string
153            if folder_path:
154                return folder_path[:-1]  # discard last "\\""
155        return None

158        self, par_name: str, case_obj_or_case_num: IntCase | int
159    ) -> Any:
160        """Get parameter value for a certain case.
161
162        Note that the values in 'parameter_values' can be
163
164        - a list/tuple where each element corresponds to a case number
165        - or a single value which is used for all cases
166
167        Args:
168            par_name (str): Parameter name

170            case_obj_or_case_num (IntCase | int): Either the case number (int) or

184            except IndexError:

208    def get_case_params_value_string(
209        self,
210        case_obj_or_case_num: IntCase | int,
211        omitted_parameters: list[str] = None,
212        delimiter: str = None,
213        equals_sign: str = None,
214        anonymous_parameters: list[str] = None,
215    ) -> str:
216        """Get parameter-value string for a case
217
218        Args:
219            case_obj_or_case_num (IntCase | int): study case
220
221            omitted_parameters (list[str], optional): parameters will not be considered. Defaults to None.
222
223            delimiter (str, optional): delimiter between parameter value pairs. Defaults to None.
224
225            equals_sign (str, optional): sign between parameter name and value. Defaults to None.
226
227            anonymous_parameters (list[str], optional): only the value of these parameters will be added (not their name). Defaults to None.
228
229        Returns:
230            str: Parameters and their values
231        """
232        case_num = self._handle_case_input(case_obj_or_case_num)
233        if not delimiter:
234            delimiter = self.delimiter
235        if not equals_sign:
236            equals_sign = " _ "
237        parameter_values_string = ""
238        for par_name in self.parameter_values.keys():
239            if omitted_parameters is None or par_name not in omitted_parameters:
240                par_value = self.get_value_of_parameter_for_case(par_name, case_num)
241                if (
242                    par_value is not None
243                    or not self.ignore_parameters_that_are_none_in_names
244                ):
245                    add_to_string = str(par_value) + delimiter
246                    if anonymous_parameters is None:
247                        if par_name not in self.anonymous_parameters:
248                            add_to_string = par_name + equals_sign + add_to_string
249                    elif par_name not in anonymous_parameters:
250                        add_to_string = par_name + equals_sign + add_to_string
251                    parameter_values_string += add_to_string
252        # discard last delimiter
253        return parameter_values_string[: -len(delimiter)]

350    def get_study_cases(
351        self,
352        conditions: dict[str, Callable] | Callable,
353        return_case_numbers: bool = False,
354    ) -> list[IntCase] | tuple[list[IntCase], list[int]]:
355        """Retrieve study case objects depending on parameter values.
356
357        Example 1:
358          get_study_cases({"par1": lambda x: x == 2, "par2": lambda x: x>0})
359            This returns the study cases for which 'par1' equals 2 and 'par2' is
360            positive.
361
362        Example 2 (lambda function):
363          get_study_cases(lambda x: x[0] >= 2 and x[2] == 'A')
364
365        Args:
366            conditions (dict[str, Callable] | Callable):
367              Either a dictionary with
368                keys: parameter names
369                values: Callables with boolean return value depending on
370                  parameter (key)
371
372              or a single Callable that accepts an iterable containing all
373                parameters.
374                Example: lambda x: x[0] >= 2 and x[2] == 'A'
375
376                Note that the order of the parameters in x must be the same as the
377                order of the keys in self.parameter_values.
378
379            return_case_numbers (bool, optional): If True, the case number are also returned. Defaults to False.
380
381        Raises:
382            ValueError: If 'callable' is invalid.
383
384        Returns:
385            list[IntCase] | tuple[list[IntCase], list[int]]: study case objects and case numbers (optional)
386        """
387        cases_objects = []
388        if return_case_numbers:
389            case_numbers = []
390        # Dictionary with conditions for parameters
391        if isinstance(conditions, dict):
392            for case_num, case_obj in enumerate(self._study_cases):
393                conditions_fullfiled = True
394                for parameter, condition in conditions.items():
395                    if not condition(
396                        self.get_value_of_parameter_for_case(parameter, case_num)
397                    ):
398                        conditions_fullfiled = False
399                        break
400                if conditions_fullfiled:
401                    cases_objects.append(case_obj)
402                    if return_case_numbers:
403                        case_numbers.append(case_num)
404        # Callables (e.g. lambda function for all parameters)
405        elif callable(conditions):
406            for case_num, case_obj in enumerate(self._study_cases):
407                parameter_values = self.get_value_of_all_parameters_for_case(case_num)
408                if conditions(parameter_values):
409                    cases_objects.append(case_obj)
410                    if return_case_numbers:
411                        case_numbers.append(case_num)
412        else:
413            raise ValueError("conditions must be a dictionary or a callable.")
414        if not return_case_numbers:
415            return cases_objects
416        else:
417            return cases_objects, case_numbers

419    def get_study_cases_from_string(
420        self, conditions: str, return_case_numbers: bool = False
421    ) -> list[IntCase] | tuple[list[IntCase], list[int]]:
422        """This method is another convenient way to get study cases according to conditions. The conditions are a simple lambda function argument string (see example below). This method is more convenient but less safe than 'self.get_study_cases' because the conditions string is evaluated and a lambda function is created from it. Using eval() statements is generally not recommended due to unforeseeable behavior.
423        However, for convenience, it is used here.
424
425        Args:
426          conditions (str):
427            lambda function argument string:
428              Example: "p HV load >= 2 and (control 1 == 'A' and control 2 != 'S')"
429
430          return_case_numbers (bool): If True, not only study case objects, but also study case numbers (indexes) are returned as a tuple.
431
432        Returns:
433            list[IntCase] | tuple[list[IntCase], list[int]]: study case objects and case numbers (optional)
434        """
435        # To create a lambda function from the conditions string, the parameter names need to be replaced by proper python variable names (e.g. "p HV load" is not a proper variable name because of the spaces). Therefore, a dict with the mapping  from parameter names to a list is required (e.g. {"p HV load": x[0],..})
436        par_name_to_list_mapping = self._get_parameter_name_to_list_mapping()
437        # Then the parameter names are replaced in conditions. Note that there could be strings inside conditions which should not be considered: For example, if there is condition "control == 'control A'", then only the parameter name 'control' should be replaced, not the value 'control A' which also contains 'control', i.e x[0] == 'control A'.
438        condition_strings_list = (
439            PFStringManipulation.replace_outside_or_inside_of_strings_in_a_string(
440                conditions, par_name_to_list_mapping
441            )
442        )
443        lambda_fun = "lambda x: " + "".join(condition_strings_list).strip()
444        # Going back to example in the docstring, we now have a lambda function string that can be executed: "lambda x: x[0] >= 2 and (x[1] == 'A' and x[2] != 'S')"
445        lambda_fun = eval(lambda_fun)
446        cases = self.get_study_cases(
447            lambda_fun, return_case_numbers=return_case_numbers
448        )
449        return cases

455    def apply_permutation(
456        self, omitted_combinations: list[dict[str, list]] = None
457    ) -> None:
458        """Replaces the values in 'parameter_values' with the permutation of
459        their unique elements.
460
461        Use this method if you want to create cases of the permutation of all
462        parameters. Note that 'parameter_values' is changed irreversibly.
463
464        Args:
465            omitted_combinations (list[dict[str, list]], optional): Parameter combinations that are omitted. Defaults to None.
466
467            Example:
468              omitted_combinations = [
469                {"Par 1": [1, 0], "Par 2": ["R", "T"]},
470                {"Par 3": [2], "Par 1": "all", "Par 4": "all"},
471              ]
472              Note that the 'all' keyword is used which means that all combiations with this parameter are omitted.
473        """
474        if omitted_combinations:
475            self.omitted_combinations = omitted_combinations
476        # Make sure values are unique
477        for param_name in self.parameter_values.keys():
478            self.parameter_values[param_name] = list(
479                set(self.parameter_values[param_name])
480            )
481        # Use 'product' to get iterable that returns permutation
482        permutation_iterable = product(*self.parameter_values.values())
483        # Clear values
484        original_parameter_values = self.parameter_values.copy()
485        for param_name in self.parameter_values.keys():
486            self.parameter_values[param_name] = []
487        # Copy values from iterable
488        for values_of_all_parameters_for_case in permutation_iterable:
489            if self.omitted_combinations:
490                values_of_all_parameters_for_case = self._filter_omitted_combinations(
491                    values_of_all_parameters_for_case, original_parameter_values
492                )
493            if values_of_all_parameters_for_case:
494                for param_num, param_name in enumerate(self.parameter_values.keys()):
495                    self.parameter_values[param_name].append(
496                        values_of_all_parameters_for_case[param_num]
497                    )

547        """Handle PF study case object (IntCase) or case number (int) input and return the case number.

660    def _filter_omitted_combinations(
661        self,
662        values_of_all_parameters_for_case: list | tuple,
663        original_parameter_values: dict[str, Any],
664    ) -> list | None:
665        """Filter the parameter combinations that should be omitted. Returns the parameters if the combination is not omitted, otherwise returns None.
666
667        The difficulty is the handling of the 'all' case. In this case, the combination with any value of a parameter is omitted, but combinations of the remaining parameters must be allowed.
668
669        Args:
670            values_of_all_parameters_for_case (list | tuple): parameter values for specific case
671
672            original_parameter_values (dict[str, Any]): original values BEFORE permutation is applied.
673
674        Returns:
675            list | None: parameters if the combination is not omitted, otherwise returns None
676        """
677        # Create list from tuple because values might be changed/set to 'None'
678        values_of_all_parameters_for_case = list(values_of_all_parameters_for_case)
679        for omitted_combination_dict in self.omitted_combinations:
680            is_omitted_combination = True
681            for param_num, param_name in enumerate(self.parameter_values.keys()):
682                if param_name in omitted_combination_dict:
683                    if omitted_combination_dict[param_name] == "all":
684                        # If the parameter is 'all', then only one case should be created. The case where the value of the parameter takes on the value of the first entry in  original_parameter_values[param_name] is selected. This must be the case for all parameters that are set to 'all'
685                        if not (
686                            original_parameter_values[param_name][0]
687                            == values_of_all_parameters_for_case[param_num]
688                        ):
689                            return False
690                        else:
691                            is_omitted_combination = False
692                            values_of_all_parameters_for_case[param_num] = None
693                    elif (
694                        not values_of_all_parameters_for_case[param_num]
695                        in omitted_combination_dict[param_name]
696                    ):
697                        is_omitted_combination = False
698                        break
699            if is_omitted_combination:
700                return False
701        return values_of_all_parameters_for_case

457        self, omitted_combinations: list[dict[str, list]] = None
458    ) -> None:
459        """Replaces the values in 'parameter_values' with the permutation of
460        their unique elements.
461
462        Use this method if you want to create cases of the permutation of all
463        parameters. Note that 'parameter_values' is changed irreversibly.
464
465        Args:
466            omitted_combinations (list[dict[str, list]], optional): Parameter combinations that are omitted. Defaults to None.
467
468            Example:
469              omitted_combinations = [
470                {"Par 1": [1, 0], "Par 2": ["R", "T"]},
471                {"Par 3": [2], "Par 1": "all", "Par 4": "all"},
472              ]
473              Note that the 'all' keyword is used which means that all combiations with this parameter are omitted.
474        """
475        if omitted_combinations:
476            self.omitted_combinations = omitted_combinations
477        # Make sure values are unique
478        for param_name in self.parameter_values.keys():
479            self.parameter_values[param_name] = list(
480                set(self.parameter_values[param_name])
481            )
482        # Use 'product' to get iterable that returns permutation
483        permutation_iterable = product(*self.parameter_values.values())
484        # Clear values
485        original_parameter_values = self.parameter_values.copy()
486        for param_name in self.parameter_values.keys():
487            self.parameter_values[param_name] = []
488        # Copy values from iterable
489        for values_of_all_parameters_for_case in permutation_iterable:
490            if self.omitted_combinations:
491                values_of_all_parameters_for_case = self._filter_omitted_combinations(
492                    values_of_all_parameters_for_case, original_parameter_values
493                )
494            if values_of_all_parameters_for_case:
495                for param_num, param_name in enumerate(self.parameter_values.keys()):
496                    self.parameter_values[param_name].append(
497                        values_of_all_parameters_for_case[param_num]

661    def _filter_omitted_combinations(
662        self,
663        values_of_all_parameters_for_case: list | tuple,
664        original_parameter_values: dict[str, Any],
665    ) -> list | None:
666        """Filter the parameter combinations that should be omitted. Returns the parameters if the combination is not omitted, otherwise returns None.
667
668        The difficulty is the handling of the 'all' case. In this case, the combination with any value of a parameter is omitted, but combinations of the remaining parameters must be allowed.
669
670        Args:
671            values_of_all_parameters_for_case (list | tuple): parameter values for specific case
672
673            original_parameter_values (dict[str, Any]): original values BEFORE permutation is applied.
674
675        Returns:
676            list | None: parameters if the combination is not omitted, otherwise returns None
677        """
678        # Create list from tuple because values might be changed/set to 'None'
679        values_of_all_parameters_for_case = list(values_of_all_parameters_for_case)
680        for omitted_combination_dict in self.omitted_combinations:
681            is_omitted_combination = True
682            for param_num, param_name in enumerate(self.parameter_values.keys()):
683                if param_name in omitted_combination_dict:
684                    if omitted_combination_dict[param_name] == "all":
685                        # If the parameter is 'all', then only one case should be created. The case where the value of the parameter takes on the value of the first entry in  original_parameter_values[param_name] is selected. This must be the case for all parameters that are set to 'all'
686                        if not (
687                            original_parameter_values[param_name][0]
688                            == values_of_all_parameters_for_case[param_num]
689                        ):
690                            return False
691                        else:
692                            is_omitted_combination = False
693                            values_of_all_parameters_for_case[param_num] = None
694                    elif (
695                        not values_of_all_parameters_for_case[param_num]
696                        in omitted_combination_dict[param_name]
697                    ):
698                        is_omitted_combination = False
699                        break
700            if is_omitted_combination:
701                return False
702        return values_of_all_parameters_for_case
