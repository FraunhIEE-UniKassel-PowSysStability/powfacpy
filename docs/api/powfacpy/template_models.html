powfacpy.template_models

Python interfaces to PowerFactory dynamic-model templates.

Each concrete class wraps an applied dynamic unit (a composite model plus its network element) that was built from a known template - from the global DIgSILENT library or a user library - and exposes information that is otherwise buried in the DSL controllers (e.g. the inertia of a grid-forming converter).

powfacpy.template_models.matcher.TemplateMatcher detects which template (if any) a composite model in a grid corresponds to and returns the matching class.

Sub-packages mirror the folder structure of the source library, e.g. digsilent_library/grid_forming_converters/ <-> Templates\Grid-forming Converters.

 90    def __init__(
 91        self,
 92        template_classes: list[type[TemplateModel]] | None = None,
 93        app=None,
 94        library_index=None,
 95    ) -> None:
 96        """
 97        Args:
 98            template_classes: the `TemplateModel` subclasses to match against;
 99                defaults to every class registered via `@register` (i.e. all of
100                `powfacpy.template_models.TEMPLATE_MODELS`).
101            app: a PowerFactory application. If given (and `library_index` is
102                not), a `LibraryTemplateIndex` is created from it so every
103                `MatchResult` also carries `library_template_paths`. Omit to
104                skip library-template detection entirely.
105            library_index: a ready `LibraryTemplateIndex` to use instead of
106                building one from `app` (e.g. to share one across matchers).
107        """
108        if template_classes is None and not TEMPLATE_MODELS:
109            # make sure the concrete classes have registered themselves
110            import powfacpy.template_models  # noqa: F401
111        self.template_classes = (
112            list(template_classes)
113            if template_classes is not None
114            else list(TEMPLATE_MODELS)
115        )
116        if library_index is None and app is not None:
117            from powfacpy.template_models.library_index import LibraryTemplateIndex
118
119            library_index = LibraryTemplateIndex(app)
120        self.library_index = library_index

122    # ------------------------------------------------------------------ #
123    def identify(
124        self, composite_model: ElmComp, level: Level = "signature"
125    ) -> MatchResult:
126        """Best match for `composite_model` among the template classes.
127
128        Args:
129            composite_model: the applied `ElmComp` to identify.
130            level: the strongest check to attempt - ``"linked"`` (frame still
131                under the global library), ``"signature"`` (a wired DSL block
132                definition identifies the template; the default), or
133                ``"strict"`` (also the frame name and parameter-name set).
134                Weaker checks are always tried too; a stronger `level` only
135                enables the extra ones.
136
137        Returns:
138            MatchResult: `template_class` is None if nothing matched. If the
139            matcher has a `library_index`, `library_template_paths` and the
140            ``"library_frame"`` evidence key are filled in regardless of the
141            class match.
142        """
143        best = MatchResult(composite_model, None, 0.0, None)
144        for template_class in self.template_classes:
145            result = self._score(composite_model, template_class, level)
146            if result.confidence > best.confidence:
147                best = result
148        if self.library_index is not None:
149            library_match = self.library_index.identify(composite_model)
150            best.library_template_paths = list(library_match.template_paths)
151            best.evidence.setdefault("library_frame", library_match.frame_name)
152        return best

157        return [self.identify(cm, level) for cm in composite_models]

160    def _score(
161        self,
162        composite_model: ElmComp,
163        template_class: type[TemplateModel],
164        level: Level,
165    ) -> MatchResult:
166        frame = composite_model.typ_id
167        wired_dsl = [
168            elm
169            for slot, elm in zip(composite_model.pblk, composite_model.pelm)
170            if elm is not None and elm.GetClassName() == "ElmDsl"
171        ]
172        wired_blkdef_names = {
173            dsl.typ_id.loc_name for dsl in wired_dsl if dsl.typ_id is not None
174        }
175        evidence: dict = {"wired_block_definitions": sorted(wired_blkdef_names)}
176
177        signature_hit = bool(
178            set(template_class.SIGNATURE_BLKDEF_NAMES) & wired_blkdef_names
179        )
180        frame_hit = (
181            frame is not None
182            and frame.loc_name in template_class.FRAME_BLKDEF_NAMES
183        )
184
185        # --- linked -------------------------------------------------------
186        linked = frame is not None and frame.GetFullName().startswith(
187            _GLOBAL_LIBRARY_PREFIX
188        )
189        if linked and signature_hit:
190            evidence["frame_full_name"] = frame.GetFullName()
191            return MatchResult(
192                composite_model, template_class, 1.0, "linked", evidence
193            )
194
195        if level == "linked":
196            return MatchResult(composite_model, None, 0.0, None, evidence)
197
198        # --- signature --------------------------------------------------
199        if not signature_hit:
200            return MatchResult(composite_model, None, 0.0, None, evidence)
201        evidence["signature_block_definitions"] = sorted(
202            set(template_class.SIGNATURE_BLKDEF_NAMES) & wired_blkdef_names
203        )
204
205        if level == "signature":
206            confidence = 0.9 if frame_hit else 0.75
207            return MatchResult(
208                composite_model, template_class, confidence, "signature", evidence
209            )
210
211        # --- strict ---------------------------------------------------
212        signature_dsl = next(
213            (
214                dsl
215                for dsl in wired_dsl
216                if dsl.typ_id is not None
217                and dsl.typ_id.loc_name in template_class.SIGNATURE_BLKDEF_NAMES
218            ),
219            None,
220        )
221        params_ok = self._parameter_names_match(signature_dsl, template_class)
222        evidence["frame_name_match"] = frame_hit
223        evidence["parameter_names_match"] = params_ok
224        if frame_hit and params_ok:
225            return MatchResult(
226                composite_model, template_class, 0.98, "strict", evidence
227            )
228        return MatchResult(composite_model, None, 0.0, None, evidence)
229
230    @staticmethod
231    def _parameter_names_match(
232        signature_dsl, template_class: type[TemplateModel]
233    ) -> bool:
234        expected = getattr(template_class, "SIGNATURE_PARAMETER_NAMES", None)

115    @cached_property
116    def _templates(self) -> list[PFGeneral]:
117        """Every `IntTemplate` under `ROOT_DISPLAY_PATH`, recursively.
118
119        The walk descends only through folders (`IntFolder` / `IntPrjfolder`),
120        never into an `IntTemplate`'s own contents, so it stays cheap.
121
122        Returns:
123            list[PFGeneral]: the `IntTemplate` objects, in library order.
124        """
125        root = self.library.get_object(self.ROOT_DISPLAY_PATH)
126        found: list[PFGeneral] = []
127
128        def walk(folder: PFGeneral) -> None:
129            for child in folder.GetContents():
130                class_name = child.GetClassName()
131                if class_name == "IntTemplate":
132                    found.append(child)
133                elif class_name in ("IntFolder", "IntPrjfolder"):
134                    walk(child)
135
136        walk(root)
137        return found

140    def frame_path_to_templates(self) -> dict[str, list[str]]:
141        """Index keyed by the frame block definition's full database path.

146
147        Returns:

155        """Index keyed by the frame block definition's ``loc_name`` only.

168    def _build_index(self, key: str) -> dict[str, list[str]]:
169        """Build one of the frame indices.
170
171        Args:
172            key: ``"full_name"`` to key by ``frame.GetFullName()``,
173                ``"loc_name"`` to key by ``frame.loc_name``.
174
175        Returns:
176            dict[str, list[str]]: template GUI paths per frame key, each list
177            sorted and de-duplicated.
178        """
179        index: dict[str, list[str]] = {}
180        for template in self._templates:
181            template_path = self.library.gui_path(template)
182            for composite_model in template.GetContents("*.ElmComp", 1):
183                frame = composite_model.typ_id
184                if frame is None:
185                    continue
186                frame_key = (
187                    frame.GetFullName() if key == "full_name" else frame.loc_name
188                )
189                paths = index.setdefault(frame_key, [])
190                if template_path not in paths:
191                    paths.append(template_path)
192        return {k: sorted(v) for k, v in index.items()}

197    def identify(self, composite_model: ElmComp) -> LibraryTemplateMatch:
198        """Best guess at the library template `composite_model` was built from.
199
200        Tries the full-path index first (`frame_path_to_templates`), then the
201        name-only fallback (`frame_name_to_templates`).
202
203        Args:
204            composite_model: an applied `ElmComp` (e.g. an `ElmGenstat`'s or
205                `ElmSym`'s control composite model in a grid).
206
207        Returns:
208            LibraryTemplateMatch: falsy (empty `template_paths`) if the frame is
209            missing or matches no library template.
210        """

212        if frame is None:
213            return LibraryTemplateMatch(composite_model)
214        by_path = self.frame_path_to_templates.get(frame.GetFullName())
215        if by_path:
