 34    def get_transformers_sharing_type(self) -> list[ElmTr2]:
 35        """Other calculation-relevant `ElmTr2` that use the same `TypTr2`.
 36
 37        Changing the type in place (e.g. its rated power) would affect all of them.
 38        """
 39        transformer_type: TypTr2 = self._obj.typ_id
 40        if transformer_type is None:
 41            return []
 42        type_name = transformer_type.GetFullName()
 43        own_name = self._obj.GetFullName()
 44        act_prj = ActiveProjectCached()
 45        return [
 46            transformer
 47            for transformer in act_prj.get_calc_relevant_obj(
 48                "*.ElmTr2", error_if_non_existent=False
 49            )
 50            if transformer.GetFullName() != own_name
 51            and transformer.typ_id is not None
 52            and transformer.typ_id.GetFullName() == type_name
 53        ]

 55    def _check_type_writable(self, copy_shared_type: bool) -> None:
 56        obj = self._obj
 57        if obj.typ_id is None:
 58            raise PFInterfaceError(
 59                f"'{obj.loc_name}' has no transformer type ('typ_id'); its type cannot be modified."
 60            )
 61        others = self.get_transformers_sharing_type()
 62        if others and not copy_shared_type:
 63            names = ", ".join(sorted(o.loc_name for o in others))
 64            raise PFInterfaceError(
 65                f"The type '{obj.typ_id.loc_name}' of '{obj.loc_name}' is shared with "
 66                f"{len(others)} other transformer(s) ({names}). Pass copy_shared_type=True "
 67                "to give this transformer a private copy of its type first."
 68            )
 69
 70    def _writable_type(self, copy_shared_type: bool) -> TypTr2:
 71        """Return the transformer's `TypTr2`, ready to be modified in place.
 72
 73        Runs `_check_type_writable`; if the type is shared and `copy_shared_type` is True, the transformer is first given a private copy of its type (next to the original) and that copy is returned and re-pointed to via `typ_id`.
 74        """
 75        self._check_type_writable(copy_shared_type)
 76        transformer_type: TypTr2 = self._obj.typ_id
 77        if self.get_transformers_sharing_type():
 78            act_prj = ActiveProjectCached()
 79            # not use_existing: several transformers can share both a type and a
 80            # loc_name (a fleet built from one template), and use_existing would
 81            # then hand them all the same "copy" - collapsing the private types
 82            # back into one. A name clash instead auto-suffixes "(1)", "(2)".
 83            transformer_type = act_prj.copy_single_obj(
 84                transformer_type,
 85                transformer_type.GetParent(),
 86                new_name=f"{transformer_type.loc_name} ({self._obj.loc_name})",
 87                overwrite=False,
 88                use_existing=False,
 89            )
 90            self._obj.typ_id = transformer_type
 91        return transformer_type

 93    def make_type_private(self) -> TypTr2:
 94        """Give the transformer a private copy of its `TypTr2` if it currently shares one.
 95
 96        Returns the transformer's type (the copy, or the original if it was already private).
 97        """
 98        return self._writable_type(copy_shared_type=True)

100    def set_rated_apparent_power(
101        self, apparent_power: float, *, copy_shared_type: bool = False
102    ) -> None:
103        """Set the rated apparent power [MVA] (`TypTr2.strn`).
104
105        The short-circuit voltage (`uktr` %) is kept, and the copper / iron losses (`pcutr`, `pfe`) are scaled with the rating, so the per-unit impedance and losses are preserved on the new rating.
106
107        Args:
108            apparent_power: new rating; `strn` is set to `apparent_power / ntnum`.
109
110            copy_shared_type: give the transformer a private copy of its type before changing the rating (required when the type is shared).
111        """
112        transformer_type = self._writable_type(copy_shared_type)
113        old = transformer_type.strn * self._obj.ntnum
114        transformer_type.strn = apparent_power / self._obj.ntnum
115        if old:
116            ratio = apparent_power / old
117            for loss_attribute in ("pcutr", "pfe"):
118                if transformer_type.HasAttribute(loss_attribute):
119                    transformer_type.SetAttribute(
120                        loss_attribute,
121                        transformer_type.GetAttribute(loss_attribute) * ratio,
122                    )
123
124    def scale_rated_apparent_power(
125        self, factor: float, *, copy_shared_type: bool = False
126    ) -> None:
127        self.set_rated_apparent_power(
128            self.rated_apparent_power * factor, copy_shared_type=copy_shared_type

171

173        """
174        transformer_type = self._obj.typ_id
175        tap_factor = 1.0 + (

178            / 100.0
179        )
