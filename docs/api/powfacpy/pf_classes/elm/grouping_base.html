 29    def add_results_variable_for_elms(
 30        self,
 31        condition_for_elms: Callable,
 32        result_variables: str | list[str],
 33        results_obj: ElmRes | None = None,
 34    ) -> None:
 35        """Add results variable for internal elements selected by 'condition'.
 36
 37        Args:
 38            condition_for_elms (Callable): Condition to select elements, e.g. 'lambda x: x.GetClassName() = "ElmTerm"'
 39
 40            result_variables (str | list[str]): Results variable name(s).
 41
 42            results_obj (ElmRes | None, optional): Results object where variables are added. Defaults to None ('get_from_study_case' is used).
 43        """
 44        act_prj = ActiveProjectCached()
 45        if not results_obj:

 47        else:
 48            results_obj = act_prj._handle_single_pf_object_or_path_input(results_obj)

 50        for elm in elms:

 53    def get_internal_elms(self, condition: Callable | None = None) -> list[PFGeneral]:
 54        """Get internal elements with optional condition.
 55
 56        Args:
 57            condition_for_elms (Callable): Condition to select elements, e.g. 'lambda x: x.GetClassName() = "ElmTerm"'
 58
 59        Returns:
 60            list[PFGeneral]: list of network elements.
 61        """
 62        elms = self.get_all_internal_elms()

 68    def get_internal_elms_of_class(
 69        self, class_name: str, condition: Callable | None = None
 70    ) -> list[PFGeneral]:
 71        """Internal elements of a given class, optionally further filtered by `condition`.

 76        return self.get_internal_elms(

 78            and (condition is None or condition(x))

 81    def get_network_elements_of_plant_models(
 82        self,
 83        elm_objs: list[str] | list[PFGeneral],
 84        condition: Callable,
 85    ) -> dict:
 86        elm_objs = list(elm_objs)
 87        if isinstance(elm_objs[0], str):
 88            objs = []
 89            for cls in elm_objs:
 90                objs += self.get_internal_elms_of_class(cls)
 91            elm_objs = objs
 92        obj_netelm = {}
 93        for obj in elm_objs:
 94            obj = convert_pf_obj_to_powfacpy(obj)
 95            netelm = obj.get_network_elements_of_plant_model(
 96                condition, error_if_non_existent=False
 97            )
 98            obj_netelm[obj._obj] = netelm
 99        return obj_netelm

101    def get_network_elements_of_plant_models_sorted_by_block_definition(
102        self,
103        elm_objs: list[str] | list[PFGeneral],
104        condition: Callable,
105    ) -> dict:
106        obj_netelm = self.get_network_elements_of_plant_models(elm_objs, condition)
107        blkdef_elm = {}
108        for obj, netelms in obj_netelm.items():
109            for netelm in netelms:
110                blkdef = netelm.typ_id
111                val = blkdef_elm.get(blkdef)
112                if not val:
113                    blkdef_elm[blkdef] = {"elm": [], "elmdsl": []}
114                blkdef_elm[blkdef]["elm"].append(obj)
115                blkdef_elm[blkdef]["elmdsl"].append(netelm)
116        return blkdef_elm

272    def is_neighbor(self, grouping: AreaOrZone | ElmAreaOrZone) -> bool:
273        grouping = self._get_elm_input(grouping)
274        interchange_flag = self._obj.CalculateInterchangeTo(grouping)
275        if interchange_flag > 0:
276            return True
277        elif interchange_flag == 0:
278            return False
279        elif self._obj == grouping:
280            raise ValueError(
281                f"Potential neighbor is equal to grouping ('{self._obj.loc_name}')"
282            )
283        else:
284            raise PFInvalidLoadFlow()
285
286    def get_neighbors(
287        self, groupings: list[AreaZoneBase] | list[ElmAreaOrZone] | None = None
288    ) -> list[AreaZoneBase] | list[ElmAreaOrZone]:
289        if groupings is None:
290            groupings = self.get_all_groupings_of_same_type()
291        groupings = [self._get_elm_input(g) for g in groupings]
292        return [g for g in groupings if not g == self._obj and self.is_neighbor(g)]

295    def load_flow_power_exchange_with_in_MVA(
296        self, grouping: ElmAreaOrZone | AreaOrZone, return_value_no_exchange=np.nan
297    ) -> complex:
298        """Get load flow apparent power exchange (in MVA) with other grouping.
299
300        Args:
301            grouping (ElmAreaOrZone | AreaOrZone): other grouping
302            return_value_no_exchange (_type_, optional): return value when there is no connection. Defaults to np.nan.
303
304        Returns:
305            complex: load flow apparent power exchange (in MVA) if connected
306        """
307        grouping = self._get_elm_input(grouping)
308        interchange_flag = self._obj.CalculateInterchangeTo(grouping)
309        if interchange_flag > 0:
310            return self._obj.GetAttribute("c:Pinter") + 1j * self._obj.GetAttribute(
311                "c:Qinter"
312            )
313        elif interchange_flag == 0:
314            return return_value_no_exchange
315        else:
316            raise PFInvalidLoadFlow()
317
318    def load_flow_total_power_loads_in_MVA(self) -> complex:

323        except AttributeError:
324            raise PFInvalidLoadFlow()
