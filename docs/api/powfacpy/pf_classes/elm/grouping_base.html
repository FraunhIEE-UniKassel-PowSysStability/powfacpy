 72        self,

 74        condition: Callable,
 75    ) -> dict:

 91    def get_network_elements_of_plant_models_sorted_by_block_definition(
 92        self,
 93        elm_objs: list[str] | list[PFGeneral],
 94        condition: Callable,
 95    ) -> dict:
 96        obj_netelm = self.get_network_elements_of_plant_models(elm_objs, condition)
 97        blkdef_elm = {}
 98        for obj, netelms in obj_netelm.items():
 99            for netelm in netelms:
100                blkdef = netelm.typ_id
101                val = blkdef_elm.get(blkdef)
102                if not val:
103                    blkdef_elm[blkdef] = {"elm": [], "elmdsl": []}
104                blkdef_elm[blkdef]["elm"].append(obj)
105                blkdef_elm[blkdef]["elmdsl"].append(netelm)
106        return blkdef_elm

202    def is_neighbor(self, grouping: AreaOrZone | ElmAreaOrZone) -> bool:
203        grouping = self._get_elm_input(grouping)
204        interchange_flag = self._obj.CalculateInterchangeTo(grouping)
205        if interchange_flag > 0:
206            return True
207        elif interchange_flag == 0:
208            return False
209        elif self._obj == grouping:
210            raise ValueError(
211                f"Potential neighbor is equal to grouping ('{self._obj.loc_name}')"
212            )
213        else:
214            raise PFInvalidLoadFlow()
215
216    def get_neighbors(
217        self, groupings: list[AreaZoneBase] | list[ElmAreaOrZone] | None = None
218    ) -> list[AreaZoneBase] | list[ElmAreaOrZone]:
219        if groupings is None:
220            groupings = self.get_all_groupings_of_same_type()
221        groupings = [self._get_elm_input(g) for g in groupings]
222        return [g for g in groupings if not g == self._obj and self.is_neighbor(g)]

225    def load_flow_power_exchange_with_in_MVA(
226        self, grouping: ElmAreaOrZone | AreaOrZone, return_value_no_exchange=np.nan
227    ) -> complex:
228        """Get load flow apparent power exchange (in MVA) with other grouping.
229
230        Args:
231            grouping (ElmAreaOrZone | AreaOrZone): other grouping
232            return_value_no_exchange (_type_, optional): return value when there is no connection. Defaults to np.nan.
233
234        Returns:
235            complex: load flow apparent power exchange (in MVA) if connected
236        """
237        grouping = self._get_elm_input(grouping)
238        interchange_flag = self._obj.CalculateInterchangeTo(grouping)
239        if interchange_flag > 0:
240            return self._obj.GetAttribute("c:Pinter") + 1j * self._obj.GetAttribute(
241                "c:Qinter"
242            )
243        elif interchange_flag == 0:
244            return return_value_no_exchange
245        else:
246            raise PFInvalidLoadFlow()
