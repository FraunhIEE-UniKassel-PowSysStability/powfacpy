 33class ResultVariablesParser:
 34    """Read (parse) the result variables of PowerFactory .IntMon objects and create enumeration classes (for code completion etc.). The classes are written to the file under 'path_python_res_var_file'. The .txt files under 'path_pf_res_var_txt_files' were created manually by copying from the PF output window when clicking on 'Variable List' in the dialogue of an .IntMon file.
 35    Please have a look at these files for further understanding.
 36    """
 37
 38    def __init__(self) -> None:
 39        self.indentation_increment: str = "  "
 40        self.path_pf_res_var_txt_files: str = r".\result_variables"
 41        self.path_python_res_var_file: str = r".\src\powfacpy\result_variables.py"
 42
 43    def create_results_variables_python_file(self):
 44        """Main method to create the results variables enumeration classes by reading from the .txt files under 'path_pf_res_var_txt_files' and writing to the python file under 'path_python_res_var_file'.
 45        In the output python file, a nested class with the hierarchy
 46        ResVar -> simulation type -> Elm class
 47        is created.
 48        """
 49        with open(self.path_python_res_var_file, "w") as out_file:
 50            module_docstring = self.get_results_variables_module_docstring()
 51            out_file.write(module_docstring)
 52            out_file.write("from aenum import Enum\n\n")
 53            out_file.write("class ResVar:\n")
 54            pf_res_var_txt_files = listdir(self.path_pf_res_var_txt_files)
 55            for (
 56                pf_sim_type,
 57                powfacpy_sim_type,
 58            ) in supported_and_unsupported_simulation_types_and_names.items():
 59                if powfacpy_sim_type:
 60                    out_file.write("\n\n")
 61                    out_file.write(
 62                        self.indentation_increment
 63                        + "class "
 64                        + powfacpy_sim_type
 65                        + ":\n"
 66                    )
 67                    for res_var_file in pf_res_var_txt_files:
 68                        with open(
 69                            self.path_pf_res_var_txt_files + "//" + res_var_file, "r"
 70                        ) as in_file:
 71                            out_file.write("\n")
 72                            elm_class = res_var_file.split(".")[0]
 73                            self.write_elm_class_in_simulation_type(
 74                                elm_class, in_file, out_file, pf_sim_type
 75                            )

 77    def get_results_variables_module_docstring(self):
 78        module_docstring = f"""\"\"\"
 79This module provides enumeration classes for results variables of 'Elm' classes (network elements). This is convenient to get results variable strings and supports code completion in your IDE. If you hover over a variable, it's unit and description will be shown.  
 80
 81This file is automatically created using the .txt files in '\{self.path_pf_res_var_txt_files}' and using the module 'results_variables_parser'. The .txt files are created by copying from the output window of PF (manually) using the printed results variables when clicking on 'Variable List' in an .IntMon file of an 'Elm' class.
 82
 83Currently, the following 'Elm' classes are supported: \n
 84"""
 85        pf_res_var_files = listdir(self.path_pf_res_var_txt_files)
 86        for res_var_file in pf_res_var_files:
 87            module_docstring += res_var_file.split(".")[0] + "\n"
 88        module_docstring += (
 89            "\nThe following simulation types/result variable types are supported:\n\n"
 90        )
 91        for (
 92            pf_simtype,
 93            powfacpy_sim_type,
 94        ) in supported_and_unsupported_simulation_types_and_names.items():
 95            if powfacpy_sim_type:
 96                module_docstring += pf_simtype + " " + powfacpy_sim_type + "\n"
 97        module_docstring += '"""\n'
 98        return module_docstring

100    def write_elm_class_in_simulation_type(
101        self, elm_class: str, in_file, out_file, pf_sim_type: str
102    ):
103        """Write an elm class to the output python file for a specific simulation type. Iterates through the lines of the .txt file until it finds the simulation type. Then writes all its variables to the python files. Breaks the loop as soon as the next simualtion type is reached."""
104        out_file.write(
105            2 * self.indentation_increment + "class " + elm_class + "(Enum):\n"
106        )
107        out_file.write(
108            3 * self.indentation_increment + "_init_ = 'value __doc__'\n\n"
109        )  # required for enumeration classes with descriptions for each attribute
110        write_variables = False
111        for line in in_file:
112            line = line.rstrip()
113            if line in supported_and_unsupported_simulation_types_and_names:
114                if line == pf_sim_type:
115                    write_variables = True
116                    used_attr = []
117                elif write_variables:
118                    break
119            if write_variables and self.is_line_with_variable(line):
120                self.write_variable_line(line, used_attr, out_file)
121
122    def is_line_with_variable(self, line):
123        """Lines with variables are identified by the colon and line length. A typical line looks like this:
124        m:I2:bus1                kA     Negative-Sequence Current, Magnitude
125        """
126        return len(line) > 1 and line[1] == ":"
127
128    def write_variable_line(self, line, used_attr, out_file):
129        """Write one variable to the output python file.
130        Splits the line (that was read in) according to two (or more) empty spaces, that separate attribute, unit (optional) and description.
131        The same attributes can occur several times in the simulation type in the .txt file, so already used attributes ('used_attr') are ignored.
132        """
133        attr_and_unit_and_description = line.split("  ")
134        attr_with_underscores_instead_of_colon = attr_and_unit_and_description[
135            0
136        ].replace(":", "_")
137        if not attr_with_underscores_instead_of_colon in used_attr:
138            used_attr.append(attr_with_underscores_instead_of_colon)
139            out_line = (
140                3 * self.indentation_increment
141                + attr_with_underscores_instead_of_colon
142                + " = "
143                + '"'
144                + attr_and_unit_and_description[0]
145                + '", "'
146            )
147            for n in attr_and_unit_and_description[1:]:
148                if n:
149                    n = n.replace('"', "''")  # replaces "\"" e.g. in "Ik\""
150                    out_line += (
151                        n.rstrip().lstrip() + ", "
152                    )  # delete leading and trailing whitespace
153            out_file.write(out_line[:-2] + '"\n')  # delete last comma
